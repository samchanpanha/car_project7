   205→const INVENTORY_PAGE_SIZE = 8;
   206→
   207→function DealerInventory() {
   208→  const t = useT();
   209→  const navigate = useAppStore((s) => s.navigate);
   210→  const [status, setStatus] = useState<string>("ALL");
   211→  const [page, setPage] = useState(1); // 1-based client-side pagination
   212→  const [busyId, setBusyId] = useState<string | null>(null);
   213→  const [editing, setEditing] = useState<TenantVehicleRowDTO | null>(null);
   214→  const [editPrice, setEditPrice] = useState("");
   215→  // Full stock editor (all info + unlimited image management) — Task 16.
   216→  const [editorId, setEditorId] = useState<string | null>(null);
   217→  const [editorOpen, setEditorOpen] = useState(false);
   218→
   219→  const url = `/api/tenant/inventory${qs({ status: status === "ALL" ? undefined : status })}`;
   220→  const { data, loading, error, refetch } = useFetch<TenantVehicleRowDTO[]>(url);
   221→
   222→  // Page the full stock list client-side; refetches keep the current page.
   223→  const totalRows = data?.length ?? 0;
   224→  const pageCount = Math.max(1, Math.ceil(totalRows / INVENTORY_PAGE_SIZE));
   225→  const currentPage = Math.min(page, pageCount);
   226→  const rows = (data ?? []).slice((currentPage - 1) * INVENTORY_PAGE_SIZE, currentPage * INVENTORY_PAGE_SIZE);
   227→
   228→  const patch = async (id: string, body: Record<string, string | number | boolean>) => {
   229→    setBusyId(id);
   230→    try {
   231→      await apiSend(`/api/tenant/inventory/${id}`, "PATCH", body);
   232→      toast.success(t("toastUpdated"));
   233→      refetch();
   234→    } catch (e) {
   235→      toast.error(isForbidden(e) ? t("toastForbidden") : isFeaturedLimit(e) ? t("toastFeaturedLimit") : errorText(e));
   236→    } finally {
   237→      setBusyId(null);
   238→    }
   239→  };
   240→
   241→  const submitPrice = async () => {
   242→    if (!editing) return;
   243→    const p = Number(editPrice);
   244→    if (!Number.isFinite(p) || p <= 0) return;
   245→    await patch(editing.id, { priceUsd: p });
   246→    setEditing(null);
   247→  };
   248→
   249→  return (
   250→    <div className="space-y-4">
   251→      <div className="flex flex-wrap items-center justify-between gap-2">
   252→        <div className="flex flex-wrap gap-1.5">
   253→          {TENANT_STATUSES.map((s) => (
   254→            <button
   255→              key={s}
   256→              onClick={() => { setStatus(s); setPage(1); }}
   257→              aria-pressed={status === s}
   258→              className={cn(
   259→                "rounded-full border px-3 py-1.5 text-xs font-medium transition-colors cursor-pointer",
   260→                status === s ? "border-amber-500 bg-amber-50 text-amber-800" : "border-slate-200 bg-white text-slate-600 hover:border-slate-300"
   261→              )}
   262→            >
   263→              {t(`vst${s}` as never)}
   264→            </button>
   265→          ))}
   266→        </div>
   267→        <Button size="sm" onClick={() => navigate({ name: "sell" })} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   268→          <Plus className="h-4 w-4 mr-1.5" aria-hidden />
   269→          {t("addListing")}
   270→        </Button>
   271→      </div>
   272→
   273→      {loading && !data ? (
   274→        <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>
   275→      ) : error && !data ? (
   276→        <FetchErrorState message={error} onRetry={refetch} />
   277→      ) : !data || data.length === 0 ? (
   278→        <EmptyState icon={Car} text={t("inventoryEmpty")} />
   279→      ) : (
   280→        <div className="space-y-2.5">
   281→          {rows.map((v) => (
   282→            <div key={v.id} className="flex items-center gap-3 rounded-xl border border-slate-200 bg-white p-3">
   283→              <div className="h-14 w-20 shrink-0 overflow-hidden rounded-lg bg-slate-100">
   284→                {v.images[0] ? (
   285→                  <img src={v.images[0]} alt="" className="h-full w-full object-cover" />
   286→                ) : (
   287→                  <Car className="mx-auto h-6 w-6 text-slate-300" aria-hidden />
   288→                )}
   289→              </div>
   290→              <div className="min-w-0 flex-1">
   291→                <p className="truncate text-sm font-medium text-slate-900">
   292→                  {v.make} {v.model} {v.variant ?? ""} {v.year}
   293→                </p>
   294→                <p className="mt-0.5 text-[11px] text-slate-400">
   295→                  {formatUsd(v.priceUsd)} · <Eye className="inline h-3 w-3" aria-hidden /> {v.views}
   296→                </p>
   297→                <div className="mt-1.5 flex items-center gap-2">
   298→                  <VehicleStatusBadge status={v.status} />
   299→                  {v.featured && <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" aria-label={t("actFeature")} />}
   300→                </div>
   301→              </div>
   302→              <DropdownMenu>
   303→                <DropdownMenuTrigger asChild>
   304→                  <Button variant="outline" size="icon" className="h-9 w-9" disabled={busyId === v.id} aria-label={`${t("actions")} ${v.make} ${v.model}`}>
   305→                    {busyId === v.id ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <MoreHorizontal className="h-4 w-4" aria-hidden />}
   306→                  </Button>
   307→                </DropdownMenuTrigger>
   308→                <DropdownMenuContent align="end" className="w-48">
   309→                  <DropdownMenuLabel>{t("actions")}</DropdownMenuLabel>
   310→                  <DropdownMenuSeparator />
   311→                  <DropdownMenuItem onClick={() => { setEditorId(v.id); setEditorOpen(true); }}>
   312→                    {t("actEditListing")}
   313→                  </DropdownMenuItem>
   314→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "PUBLISHED" })}>{t("actPublish")}</DropdownMenuItem>
   315→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "DRAFT" })}>{t("actUnpublish")}</DropdownMenuItem>
   316→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "SOLD" })}>{t("actMarkSold")}</DropdownMenuItem>
   317→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "ARCHIVED" })}>{t("actArchive")}</DropdownMenuItem>
   318→                  <DropdownMenuSeparator />
   319→                  <DropdownMenuItem onClick={() => patch(v.id, { featured: !v.featured })}>
   320→                    {v.featured ? t("actUnfeature") : t("actFeature")}
   321→                  </DropdownMenuItem>
   322→                  <DropdownMenuItem onClick={() => { setEditing(v); setEditPrice(String(v.priceUsd)); }}>
   323→                    {t("actEditPrice")}
   324→                  </DropdownMenuItem>
   325→                </DropdownMenuContent>
   326→              </DropdownMenu>
   327→            </div>
   328→          ))}
   329→        </div>
   330→      )}
   331→
   332→      {/* Pagination — paged over the full stock list */}
   333→      {data && data.length > INVENTORY_PAGE_SIZE && (
   334→        <div className="space-y-1.5 pt-1">
   335→          <p className="text-center text-xs text-slate-400">
   336→            {t("paginationShowing", {
   337→              from: (currentPage - 1) * INVENTORY_PAGE_SIZE + 1,
   338→              to: Math.min(currentPage * INVENTORY_PAGE_SIZE, data.length),
   339→              n: data.length,
   340→            })}
   341→          </p>
   342→          <Pagination page={currentPage} pageCount={pageCount} onPageChange={setPage} disabled={busyId !== null} />
   343→        </div>
   344→      )}
   345→
   346→      {/* Edit price dialog */}
   347→      <Dialog open={Boolean(editing)} onOpenChange={(o) => !o && setEditing(null)}>
   348→        <DialogContent className="sm:max-w-xs">
   349→          <DialogHeader>
   350→            <DialogTitle>{t("editPriceTitle")}</DialogTitle>
   351→          </DialogHeader>
   352→          <div>
   353→            <Label htmlFor="inv-price" className="text-xs text-slate-500">{editing ? `${editing.make} ${editing.model} ${editing.year}` : ""}</Label>
   354→            <Input
   355→              id="inv-price" type="number" min={0} className="mt-1.5 h-9" value={editPrice}
   356→              onChange={(e) => setEditPrice(e.target.value)} autoFocus
   357→            />
   358→          </div>
   359→          <DialogFooter>
   360→            <Button variant="outline" size="sm" onClick={() => setEditing(null)}>{t("cancel")}</Button>
   361→            <Button size="sm" onClick={submitPrice} disabled={busyId !== null} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   362→              {t("saveChanges")}
   363→            </Button>
   364→          </DialogFooter>
   365→        </DialogContent>
   366→      </Dialog>
   367→
   368→      {/* Full stock editor — all listing info + unlimited photo manager */}
   369→      <ListingEditorDialog
   370→        vehicleId={editorId}
   371→        open={editorOpen}
   372→        onOpenChange={(o) => {
   373→          setEditorOpen(o);
   374→          if (!o) refetch();
   375→        }}
   376→      />
   377→    </div>
   378→  );
   379→}

    52→/** Read a file as a raw data URL — used for PDFs (no canvas downscale). */
    53→function rawFileToDataUrl(file: File): Promise<string> {
    54→  return new Promise((resolve, reject) => {
    55→    const reader = new FileReader();
    56→    reader.onload = () => resolve(String(reader.result));
    57→    reader.onerror = () => reject(new Error("read-failed"));
    58→    reader.readAsDataURL(file);
    59→  });
    60→}

    62→const isPdfUrl = (url: string) => url.toLowerCase().endsWith(".pdf");

    64→/**
    65→ * Optional vehicle-papers manager: one row per document (type + status +

    70→  documents,
    71→  onChange,

    81→  const inputRef = useRef<HTMLInputElement | null>(null);
    82→  // Which row is currently uploading its file (keyed like the row list).
    83→  const [uploadingKey, setUploadingKey] = useState<string | null>(null);
    84→  // The row the shared file picker was opened for.
    85→  const pendingKeyRef = useRef<string | null>(null);
    86→
    87→  const rowKey = (d: DocumentRow, i: number) => d.id ?? `new-${i}`;
    88→
    89→  const update = (key: string, patch: Partial<DocumentRow>) =>
    90→    onChange(documents.map((d, i) => (rowKey(d, i) === key ? { ...d, ...patch } : d)));
    91→
    92→  const removeAt = (idx: number) => onChange(documents.filter((_, i) => i !== idx));
    93→
    94→  const addRow = () =>
    95→    onChange([...documents, { type: "REGISTRATION_BOOK", status: "PROVIDED", notes: "", fileUrl: null }]);
    96→
    97→  const uploadFile = async (key: string, file: File) => {
    98→    const isPdf = file.type === "application/pdf" || file.name.toLowerCase().endsWith(".pdf");
    99→    const isImage = file.type.startsWith("image/");
   100→    if (!isPdf && !isImage) {
   101→      toast.warning(t("docUnsupported"));
   102→      return;
   103→    }
   104→    // PDFs cannot be downscaled client-side — enforce the server cap early.
   105→    if (file.size > 3 * 1024 * 1024) {
   106→      toast.warning(t("docTooLarge"));
   107→      return;
   108→    }
   109→    setUploadingKey(key);
   110→    try {
   111→      const dataUrl = isPdf ? await rawFileToDataUrl(file) : await fileToDataUrl(file, 1600, 0.85);
   112→      const res = await apiSend<{ url: string }>("/api/uploads", "POST", { dataUrl });
   113→      update(key, { fileUrl: res.url });
   114→    } catch {
   115→      toast.error(t("uploadFailedCount", { n: 1 }));
   116→    } finally {
   117→      setUploadingKey(null);
   118→    }
   119→  };
   120→
   121→  const statusOptions = (canVerify ? ALL_DOC_STATUSES : SELLER_DOC_STATUSES).map((s) => ({
   122→    value: s,
   123→    label: t(`doc${s}` as never),
   124→    disabled: false,
   125→  }));
   126→

   128→    <div>
   129→      {/* One shared file picker — pendingKeyRef routes the chosen file to the
   130→          row whose "Upload file" button was clicked. */}
   131→      <input
   132→        ref={inputRef}
   133→        type="file"
   134→        accept="image/*,application/pdf"
   135→        className="hidden"
   136→        aria-label={t("docUploadFile")}
   137→        onChange={(e) => {
   138→          const key = pendingKeyRef.current;
   139→          const f = e.target.files?.[0];
   140→          if (f && key) void uploadFile(key, f);
   141→          e.target.value = "";
   142→        }}
   143→      />
   144→      {documents.length > 0 && (
   145→        <div className="space-y-2">
   146→          {documents.map((d, idx) => {
   147→            const key = rowKey(d, idx);
   148→            // Sellers see admin-set VERIFIED/REJECTED as read-only options.
   149→            const options =
   150→              !canVerify && !SELLER_DOC_STATUSES.includes(d.status)
   151→                ? [...statusOptions, { value: d.status, label: t(`doc${d.status}` as never), disabled: true }]
   152→                : statusOptions;
   153→            return (
   154→              <div key={key} className="rounded-xl border border-slate-200 bg-slate-50/60 p-3">
   155→                <div className="flex items-start gap-2">
   156→                  <div className="min-w-0 flex-1">
   157→                    <Select value={d.type} onValueChange={(v) => update(key, { type: v })}>
   158→                      <SelectTrigger className="h-9 w-full" aria-label={t("docTypeLabel")}>
   159→                        <SelectValue placeholder={t("docTypeLabel")} />
   160→                      </SelectTrigger>
   161→                      <SelectContent>
   162→                        {DOC_TYPES.map((tp) => (
   163→                          <SelectItem key={tp} value={tp}>
   164→                            {t(`doc${tp}` as never)}
   165→                          </SelectItem>
   166→                        ))}
   167→                      </SelectContent>
   168→                    </Select>
   169→                  </div>
   170→                  <div className="w-32 shrink-0">
   171→                    <Select value={d.status} onValueChange={(v) => update(key, { status: v })}>
   172→                      <SelectTrigger className="h-9 w-full" aria-label={t("docStatusLabel")}>
   173→                        <SelectValue placeholder={t("docStatusLabel")} />
   174→                      </SelectTrigger>
   175→                      <SelectContent>
   176→                        {options.map((o) => (
   177→                          <SelectItem key={o.value} value={o.value} disabled={o.disabled}>
   178→                            {o.label}
   179→                          </SelectItem>
   180→                        ))}
   181→                      </SelectContent>
   182→                    </Select>
   183→                  </div>
   184→                  <button
   185→                    type="button"
   186→                    onClick={() => removeAt(idx)}
   187→                    disabled={disabled}
   188→                    className="flex h-9 w-9 shrink-0 cursor-pointer items-center justify-center rounded-lg border border-slate-200 text-slate-400 transition hover:border-red-200 hover:bg-red-50 hover:text-red-600"
   189→                    aria-label={t("removeDocument")}
   190→                  >
   191→                    <X className="h-4 w-4" aria-hidden />
   192→                  </button>
   193→                </div>
   194→
   195→                <Input
   196→                  value={d.notes}
   197→                  onChange={(e) => update(key, { notes: e.target.value })}
   198→                  placeholder={t("docNotesLabel")}
   199→                  className="mt-2 h-9 text-xs"
   200→                  aria-label={t("docNotesLabel")}
   201→                />
   202→
   203→                <div className="mt-2 flex flex-wrap items-center gap-2">
   204→                  {d.fileUrl ? (
   205→                    <>
   206→                      {isPdfUrl(d.fileUrl) ? (
   207→                        <a
   208→                          href={d.fileUrl}
   209→                          target="_blank"
   210→                          rel="noopener noreferrer"

   212→                        >
   213→                          <FileText className="h-4 w-4 text-rose-600" aria-hidden />

   215→                        </a>
   216→                      ) : (
   217→                        <a
   218→                          href={d.fileUrl}
   219→                          target="_blank"
   220→                          rel="noopener noreferrer"
   221→                          className="block overflow-hidden rounded-lg border border-slate-200"
   222→                          aria-label={t("docViewFile")}
   223→                        >
   224→                          <img src={d.fileUrl} alt="" className="h-10 w-14 object-cover" loading="lazy" />
   225→                        </a>
   226→                      )}
   227→                      <button
   228→                        type="button"
   229→                        onClick={() => update(key, { fileUrl: null })}

   232→                      >
   233→                        {t("docRemoveFile")}

   237→                    <Button
   238→                      type="button"
   239→                      variant="outline"
   240→                      size="sm"
   241→                      className="h-8"
   242→                      disabled={disabled || uploadingKey === key}
   243→                      onClick={() => {
   244→                        pendingKeyRef.current = key;
   245→                        inputRef.current?.click();
   246→                      }}
   247→                    >
   248→                      {uploadingKey === key ? (
   249→                        <>
   250→                          <Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" aria-hidden />
   251→                          {t("loading")}

   253→                      ) : (
   254→                        <>
   255→                          <FileUp className="h-3.5 w-3.5 mr-1.5" aria-hidden />

   260→                  )}
   261→                  <span className="text-[11px] text-slate-400">{t("docFileHint")}</span>
   262→                </div>
   263→              </div>
   264→            );
   265→          })}
   266→        </div>

   268→
   269→      <Button
   270→        type="button"
   271→        variant="outline"

   286→/**
   287→ * Unlimited photo manager: multi-file upload (client-downscaled, stored via
   288→ * POST /api/uploads), link adds, remove, set-as-cover. `images` is the ordered
   289→ * desired list — first entry is the cover.
   290→ */
   291→export function PhotoUploader({
   292→  images,
   293→  onChange,
   294→  disabled,
   295→}: {
   296→  images: string[];
   297→  onChange: (next: string[]) => void;
   298→  disabled?: boolean;
   299→}) {
   300→  const t = useT();
   301→  const inputRef = useRef<HTMLInputElement>(null);
   302→  const [uploading, setUploading] = useState(0);
   303→  const [urlInput, setUrlInput] = useState("");
   304→  // Latest list ref so in-flight uploads never resurrect a just-removed photo.
   305→  const latest = useRef(images);
   306→  latest.current = images;
   307→
   308→  const commit = (next: string[]) => onChange(Array.from(new Set(next)));
   309→
   310→  const uploadFiles = async (files: File[]) => {
   311→    const list = files.filter((f) => f.type.startsWith("image/"));
   312→    if (list.length === 0) return;
   313→    setUploading(list.length);
   314→    let failed = 0;
   315→    const added: string[] = [];
   316→    for (const file of list) {
   317→      try {
   318→        const dataUrl = await fileToDataUrl(file, 1280, 0.82);
   319→        const res = await apiSend<{ url: string }>("/api/uploads", "POST", { dataUrl });
   320→        added.push(res.url);
   321→      } catch {
   322→        failed += 1;
   323→      } finally {
   324→        setUploading((n) => Math.max(0, n - 1));
   325→      }
   326→    }
   327→    if (added.length > 0) commit([...latest.current, ...added]);
   328→    if (failed > 0) toast.error(t("uploadFailedCount", { n: failed }));
   329→  };
   330→
   331→  const addUrl = () => {
   332→    const u = urlInput.trim();
   333→    if (!u || !/^(https?:\/\/|\/api\/uploads\/)/.test(u)) {
   334→      toast.warning(t("invalidPhotoUrl"));
   335→      return;
   336→    }
   337→    if (!latest.current.includes(u)) commit([...latest.current, u]);
   338→    setUrlInput("");
   339→  };
   340→
   341→  const removeAt = (idx: number) => commit(images.filter((_, i) => i !== idx));
   342→  const makeCover = (idx: number) => {
   343→    if (idx === 0) return;
   344→    commit([images[idx], ...images.filter((_, i) => i !== idx)]);
   345→  };
   346→
   347→  return (
   348→    <div>
   349→      <div className="flex flex-wrap items-center gap-2">
   350→        <input
   351→          ref={inputRef}
   352→          type="file"
   353→          accept="image/*"
   354→          multiple
   355→          className="hidden"
   356→          aria-label={t("uploadPhotos")}
   357→          onChange={(e) => {
   358→            if (e.target.files) void uploadFiles(Array.from(e.target.files));
   359→            e.target.value = "";
   360→          }}
   361→        />
   362→        <Button
   363→          type="button"
   364→          variant="outline"
   365→          size="sm"
   366→          className="h-9 border-amber-300 text-amber-800 hover:bg-amber-50 hover:text-amber-900"
   367→          disabled={disabled || uploading > 0}
   368→          onClick={() => inputRef.current?.click()}
   369→        >
   370→          {uploading > 0 ? (
   371→            <>
   372→              <Loader2 className="h-4 w-4 mr-1.5 animate-spin" aria-hidden />
   373→              {t("uploadingPhotos", { n: uploading })}
   374→            </>
   375→          ) : (
   376→            <>
   377→              <ImagePlus className="h-4 w-4 mr-1.5" aria-hidden />
   378→              {t("uploadPhotos")}
   379→            </>
   380→          )}
   381→        </Button>
   382→        <span className="text-[11px] text-slate-400">{t("photosUnlimitedHint")}</span>
   383→      </div>
   384→
   385→      <div className="mt-2 flex gap-2">
   386→        <Input
   387→          value={urlInput}
   388→          onChange={(e) => setUrlInput(e.target.value)}
   389→          onKeyDown={(e) => {
   390→            if (e.key === "Enter") {
   391→              e.preventDefault();
   392→              addUrl();
   393→            }
   394→          }}
   395→          placeholder="https://…"
   396→          className="h-9 text-xs"
   397→          aria-label={t("addPhotoUrl")}
   398→        />
   399→        <Button type="button" variant="outline" size="sm" className="h-9" disabled={disabled} onClick={addUrl}>
   400→          <Link2 className="h-4 w-4" aria-hidden />
   401→        </Button>
   402→      </div>
   403→
   404→      {images.length > 0 && (
   405→        <div className="mt-3 grid max-h-72 grid-cols-3 gap-2 overflow-y-auto pr-1 sm:grid-cols-4">
   406→          {images.map((url, idx) => (
   407→            <div
   408→              key={`${url}-${idx}`}
   409→              className={cn(
   410→                "group relative aspect-[4/3] overflow-hidden rounded-lg border-2 bg-slate-100",
   411→                idx === 0 ? "border-amber-500" : "border-transparent",
   412→              )}
   413→            >
   414→              <img src={url} alt="" className="h-full w-full object-cover" loading="lazy" />
   415→              {idx === 0 ? (
   416→                <span className="absolute left-1 top-1 rounded-full bg-amber-500 px-1.5 py-0.5 text-[9px] font-bold text-slate-950">
   417→                  {t("coverBadge")}
   418→                </span>
   419→              ) : (
   420→                <button
   421→                  type="button"
   422→                  onClick={() => makeCover(idx)}
   423→                  disabled={disabled}
   424→                  className="absolute left-1 top-1 flex h-6 w-6 cursor-pointer items-center justify-center rounded-full bg-slate-950/60 text-white opacity-80 transition hover:bg-amber-500 hover:text-slate-950 hover:opacity-100"
   425→                  aria-label={t("setCover")}
   426→                >
   427→                  <Star className="h-3.5 w-3.5" aria-hidden />
   428→                </button>
   429→              )}
   430→              <button
   431→                type="button"
   432→                onClick={() => removeAt(idx)}
   433→                disabled={disabled}
   434→                className="absolute right-1 top-1 flex h-6 w-6 cursor-pointer items-center justify-center rounded-full bg-slate-950/60 text-white transition hover:bg-red-500"
   435→                aria-label={t("removePhoto")}
   436→              >
   437→                <X className="h-3.5 w-3.5" aria-hidden />
   438→              </button>
   439→            </div>
   440→          ))}
   441→        </div>
   442→      )}
   443→    </div>
   444→  );
   445→}

   513→export function ListingEditorDialog({
   514→  vehicleId,
   515→  open,
   516→  onOpenChange,
   517→  onSaved,
   518→}: {
   519→  vehicleId: string | null;
   520→  open: boolean;
   521→  onOpenChange: (open: boolean) => void;
   522→  onSaved?: () => void;
   523→}) {
   524→  const t = useT();
   525→  const locale = useAppStore((s) => s.locale);
   526→  const { data: meta } = useFetch<MetaDTO>(open ? "/api/meta" : null);
   527→  // GET /api/vehicles/[id] returns { vehicle, similar } — unwrap the vehicle.
   528→  const detail = useFetch<{ vehicle: ManageableVehicleDetailDTO; similar: unknown }>(
   529→    open && vehicleId ? `/api/vehicles/${vehicleId}` : null,
   530→  );
   531→  const vehicle = detail.data?.vehicle ?? null;
   532→  const [form, setForm] = useState<EditorForm | null>(null);
   533→  const [saving, setSaving] = useState(false);
   534→
   535→  useEffect(() => {
   536→    if (vehicle) setForm(toForm(vehicle));
   537→    if (detail.error) setForm(null);
   538→  }, [vehicle, detail.error]);
   539→
   540→  // Reset local state whenever a different vehicle is opened.
   541→  useEffect(() => {
   542→    setForm(null);
   543→  }, [vehicleId]);
   544→
   545→  const set = (patch: Partial<EditorForm>) => setForm((f) => (f ? { ...f, ...patch } : f));
   546→
   547→  const title = useMemo(
   548→    () => (form ? `${form.make} ${form.model} ${form.variant ?? ""} ${form.year}`.trim() : ""),
   549→    [form],
   550→  );
   551→
   552→  const validate = (): string | null => {
   553→    if (!form) return null;
   554→    if (!form.make.trim()) return locale === "km" ? "សូមបញ្ចូលម៉ាកឡាន" : "Please enter the make";
   555→    if (!form.model.trim()) return locale === "km" ? "សូមបញ្ចូលម៉ូដែល" : "Please enter the model";
   556→    const y = Number(form.year);
   557→    if (!Number.isInteger(y) || y < 1990 || y > 2026)
   558→      return locale === "km" ? "ឆ្នាំមិនត្រឹមត្រូវ" : "Enter a valid year (1990–2026)";
   559→    const p = Number(form.priceUsd);
   560→    if (!Number.isFinite(p) || p <= 0) return locale === "km" ? "សូមបញ្ចូលតម្លៃ" : "Please enter a valid price";
   561→    const m = Number(form.mileageKm);
   562→    if (Number.isNaN(m) || m < 0) return locale === "km" ? "ចម្ងាយដំណើរមិនត្រឹមត្រូវ" : "Enter a valid mileage";
   563→    if (!form.bodyType || !form.fuel || !form.transmission)
   564→      return locale === "km" ? "សូមជ្រើសប្រភេទឡាន ឥន្ធនៈ ប៉ុស្តិ៍" : "Select body type, fuel and transmission";
   565→    return null;
   566→  };
   567→
   568→  const save = async () => {
   569→    if (!form || !vehicleId) return;
   570→    const err = validate();
   571→    if (err) {
   572→      toast.warning(err);
   573→      return;
   574→    }
   575→    setSaving(true);
   576→    try {
   577→      await apiSend(`/api/vehicles/${vehicleId}`, "PATCH", {
   578→        make: form.make.trim(),
   579→        model: form.model.trim(),
   580→        variant: form.variant.trim() || null,
   581→        year: Number(form.year),
   582→        priceUsd: Number(form.priceUsd),
   583→        mileageKm: Math.round(Number(form.mileageKm) || 0),
   584→        transmission: form.transmission,
   585→        fuel: form.fuel,
   586→        engineCc: form.engineCc ? Number(form.engineCc) : null,
   587→        drivetrain: form.drivetrain || null,
   588→        bodyType: form.bodyType,
   589→        exteriorColor: form.exteriorColor.trim() || null,
   590→        city: form.city.trim() || "Phnom Penh",
   591→        district: form.district.trim() || null,
   592→        condition: form.condition,
   593→        accidentDeclared: form.accidentDeclared,
   594→        ownershipCount: form.ownershipCount ? Number(form.ownershipCount) : null,
   595→        vinLast6: form.vinLast6.trim().slice(0, 6) || null,
   596→        plateNo: form.plateNo.trim() || null,
   597→        importOrigin: form.importOrigin || null,
   598→        description: form.description.trim() || null,
   599→        descriptionKm: form.descriptionKm.trim() || null,
   600→        status: form.status,
   601→        images: form.images, // full ordered list — add/remove/reorder in one call
   602→        documents: form.documents.map((d) => ({
   603→          ...(d.id ? { id: d.id } : {}),
   604→          type: d.type,
   605→          status: d.status,
   606→          notes: d.notes.trim() || null,
   607→          fileUrl: d.fileUrl,
   608→        })),
   609→      });
   610→      toast.success(t("listingUpdated"));
   611→      onOpenChange(false);
   612→      onSaved?.();
   613→    } catch (e) {
   614→      toast.error(e instanceof Error ? e.message : t("errorGeneric"));
   615→    } finally {
   616→      setSaving(false);
   617→    }
   618→  };
   619→
   620→  return (
   621→    <Dialog open={open} onOpenChange={(o) => { if (!o && !saving) onOpenChange(false); }}>
   622→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   623→        <DialogHeader>
   624→          <DialogTitle className="flex items-center gap-2 text-base">
   625→            <CarFront className="h-4 w-4 text-amber-600" aria-hidden />
   626→            {t("editListingTitle")}
   627→          </DialogTitle>
   628→        </DialogHeader>
   629→
   630→        {detail.loading && !form ? (
   631→          <div className="space-y-3">
   632→            <Skeleton className="h-24 w-full rounded-xl" />
   633→            <Skeleton className="h-40 w-full rounded-xl" />
   634→          </div>
   635→        ) : !form ? (
   636→          <p className="py-6 text-center text-sm text-slate-500">{detail.error || t("errorGeneric")}</p>
   637→        ) : vehicle?.manage === false ? (
   638→          <p className="py-6 text-center text-sm text-slate-500">{t("errorGeneric")}</p>
   639→        ) : (
   640→          <div className="space-y-5">
   641→            {/* Photos — unlimited stock manager */}
   642→            <section>
   643→              <Label className="text-xs font-medium text-slate-500">{t("formPhotos")}</Label>
   644→              <div className="mt-2">
   645→                <PhotoUploader images={form.images} onChange={(images) => set({ images })} disabled={saving} />
   646→              </div>
   647→              <p className="mt-1.5 text-[11px] text-slate-400">
   648→                {t("photosCount", { n: form.images.length })} · {t("coverHint")}
   649→              </p>
   650→            </section>
   651→
   652→            {/* Documents — optional vehicle papers (Task 17) */}
   653→            <section>
   654→              <div className="flex items-center justify-between">
   655→                <p className="text-xs font-semibold uppercase tracking-wide text-slate-400">{t("sectionDocuments")}</p>
   656→                {form.documents.length > 0 && (
   657→                  <span className="text-[11px] text-slate-400">{t("documentsCount", { n: form.documents.length })}</span>
   658→                )}
   659→              </div>
   660→              <p className="mt-1 text-[11px] text-slate-400">{t("docsOptionalHint")}</p>
   661→              <div className="mt-2">
   662→                <DocumentsEditor
   663→                  documents={form.documents}
   664→                  onChange={(documents) => set({ documents })}
   665→                  disabled={saving}
   666→                  canVerify={vehicle.canVerifyDocuments ?? false}
   667→                />
   668→              </div>
   669→            </section>
   670→
   671→            {/* Car details */}
   672→            <section>
   673→              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-400">{t("sectionCarDetails")}</p>
   674→              <div className="grid grid-cols-2 gap-3">
   675→                <EditField label={t("formMake")}>
   676→                  <Input value={form.make} onChange={(e) => set({ make: e.target.value })} list="kl-ed-makes" />
   677→                  <datalist id="kl-ed-makes">
   678→                    {(meta?.makes ?? []).map((m) => <option key={m} value={m} />)}
   679→                  </datalist>
   680→                </EditField>
   681→                <EditField label={t("formModel")}>
   682→                  <Input value={form.model} onChange={(e) => set({ model: e.target.value })} />
   683→                </EditField>
   684→                <EditField label={t("formVariant")}>
   685→                  <Input value={form.variant} onChange={(e) => set({ variant: e.target.value })} />
   686→                </EditField>
   687→                <EditField label={t("formYear")}>
   688→                  <Input type="number" min={1990} max={2026} value={form.year} onChange={(e) => set({ year: e.target.value })} />
   689→                </EditField>
   690→                <EditField label={t("specMileage")}>
   691→                  <Input type="number" min={0} value={form.mileageKm} onChange={(e) => set({ mileageKm: e.target.value })} />
   692→                </EditField>
   693→                <EditField label={t("formColor")}>
   694→                  <Input value={form.exteriorColor} onChange={(e) => set({ exteriorColor: e.target.value })} placeholder="White" />
   695→                </EditField>
   696→                <EditField label={t("formBodyType")}>
   697→                  <EditSelect value={form.bodyType} onChange={(v) => set({ bodyType: v })} options={BODY_TYPES.map((b) => ({ value: b, label: b }))} placeholder={t("formBodyType")} />
   698→                </EditField>
   699→                <EditField label={t("formFuel")}>
   700→                  <EditSelect value={form.fuel} onChange={(v) => set({ fuel: v })} options={FUELS.map((f) => ({ value: f, label: fuelLabels[f]?.[locale] ?? f }))} placeholder={t("formFuel")} />
   701→                </EditField>
   702→                <EditField label={t("formTransmission")}>
   703→                  <EditSelect value={form.transmission} onChange={(v) => set({ transmission: v })} options={TRANSMISSIONS.map((tr) => ({ value: tr, label: transmissionLabels[tr]?.[locale] ?? tr }))} placeholder={t("formTransmission")} />
   704→                </EditField>
   705→                <EditField label={t("formDrivetrain")}>
   706→                  <EditSelect value={form.drivetrain} onChange={(v) => set({ drivetrain: v })} options={DRIVETRAINS.map((d) => ({ value: d, label: d }))} placeholder="—" />
   707→                </EditField>
   708→                <EditField label={t("formEngine")}>
   709→                  <Input type="number" min={0} value={form.engineCc} onChange={(e) => set({ engineCc: e.target.value })} />
   710→                </EditField>
   711→                <EditField label={t("formCondition")}>
   712→                  <EditSelect value={form.condition} onChange={(v) => set({ condition: v })} options={[{ value: "LIKE_NEW", label: t("condLIKE_NEW") }, { value: "USED", label: t("condUSED") }]} placeholder={t("formCondition")} />
   713→                </EditField>
   714→                <EditField label={t("formAccident")}>
   715→                  <EditSelect value={form.accidentDeclared} onChange={(v) => set({ accidentDeclared: v })} options={ACCIDENTS.map((a) => ({ value: a, label: t(`accident${a}` as never) }))} placeholder={t("formAccident")} />
   716→                </EditField>
   717→                <EditField label={t("formOwners")}>
   718→                  <Input type="number" min={0} value={form.ownershipCount} onChange={(e) => set({ ownershipCount: e.target.value })} />
   719→                </EditField>
   720→                <EditField label={t("formVin")}>
   721→                  <Input value={form.vinLast6} maxLength={6} onChange={(e) => set({ vinLast6: e.target.value.toUpperCase() })} />
   722→                </EditField>
   723→                <EditField label={t("formPlate")}>
   724→                  <Input value={form.plateNo} onChange={(e) => set({ plateNo: e.target.value })} />
   725→                </EditField>
   726→                <EditField label={t("formImport")}>
   727→                  <EditSelect value={form.importOrigin} onChange={(v) => set({ importOrigin: v })} options={IMPORTS.map((i) => ({ value: i, label: i }))} placeholder="—" />
   728→                </EditField>
   729→              </div>
   730→            </section>
   731→
   732→            {/* Price & location */}
   733→            <section>
   734→              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-400">{t("sectionPriceLocation")}</p>
   735→              <div className="grid grid-cols-2 gap-3">
   736→                <EditField label={t("formPrice")}>
   737→                  <Input type="number" min={0} value={form.priceUsd} onChange={(e) => set({ priceUsd: e.target.value })} />
   738→                </EditField>
   739→                <EditField label={t("formCity")}>
   740→                  <Input value={form.city} onChange={(e) => set({ city: e.target.value })} />
   741→                </EditField>
   742→                <EditField label={t("formDistrict")}>
   743→                  <Input value={form.district} onChange={(e) => set({ district: e.target.value })} />
   744→                </EditField>
   745→                <EditField label={t("sectionStatus")}>
   746→                  <EditSelect
   747→                    value={form.status}
   748→                    onChange={(v) => set({ status: v })}
   749→                    options={EDIT_STATUSES.map((s) => ({ value: s, label: t(`vst${s}` as never) }))}
   750→                    placeholder={t("sectionStatus")}
   751→                  />
   752→                </EditField>
   753→              </div>
   754→            </section>
   755→
   756→            {/* Description */}
   757→            <section>
   758→              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-400">{t("sectionDescription")}</p>
   759→              <div className="space-y-3">
   760→                <EditField label={t("formDescription")}>
   761→                  <Textarea rows={3} value={form.description} onChange={(e) => set({ description: e.target.value })} />
   762→                </EditField>
   763→                <EditField label={t("formDescriptionKm")}>
   764→                  <Textarea rows={3} value={form.descriptionKm} onChange={(e) => set({ descriptionKm: e.target.value })} className="font-khmer" />
   765→                </EditField>
   766→              </div>
   767→            </section>
   768→
   769→            <div className="flex items-start gap-2 rounded-lg border border-amber-200 bg-amber-50 p-3">
   770→              <TriangleAlert className="mt-0.5 h-4 w-4 shrink-0 text-amber-600" aria-hidden />
   771→              <p className="text-xs leading-relaxed text-amber-800">{t("editListingNote")}</p>
   772→            </div>
   773→          </div>
   774→        )}
   775→
   776→        <DialogFooter>
   777→          <Button variant="outline" size="sm" onClick={() => onOpenChange(false)} disabled={saving}>
   778→            {t("cancel")}
   779→          </Button>
   780→          <Button
   781→            size="sm"
   782→            onClick={save}
   783→            disabled={saving || !form || detail.loading}
   784→            className="bg-amber-500 font-semibold text-slate-950 hover:bg-amber-400"
   785→          >
   786→            {saving ? (
   787→              <>
   788→                <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden />
   789→                {t("loading")}
   790→              </>
   791→            ) : (
   792→              t("saveChanges")
   793→            )}
   794→          </Button>
   795→        </DialogFooter>
   796→      </DialogContent>
   797→    </Dialog>
   798→  );
   799→}
   800→
   801→function EditField({ label, children }: { label: string; children: React.ReactNode }) {
   802→  return (
   803→    <div>
   804→      <Label className="text-xs font-medium text-slate-500">{label}</Label>
   805→      <div className="mt-1.5">{children}</div>
   806→    </div>
   807→  );
   808→}
   809→
   810→function EditSelect({
   811→  value,
   812→  onChange,
   813→  options,
   814→  placeholder,
   815→}: {
   816→  value: string;
   817→  onChange: (v: string) => void;
   818→  options: { value: string; label: string }[];
   819→  placeholder?: string;
   820→}) {
   821→  return (
   822→    <Select value={value || undefined} onValueChange={onChange}>
   823→      <SelectTrigger className="h-9 w-full" aria-label={placeholder}>
   824→        <SelectValue placeholder={placeholder} />
   825→      </SelectTrigger>
   826→      <SelectContent>
   827→        {options.map((o) => (
   828→          <SelectItem key={o.value} value={o.value}>
   829→            {o.label}
   830→          </SelectItem>
   831→        ))}
   832→      </SelectContent>
   833→    </Select>
   834→  );
   835→}

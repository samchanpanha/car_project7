   202→
   203→function DealerInventory() {
   204→  const t = useT();
   205→  const navigate = useAppStore((s) => s.navigate);
   206→  const [status, setStatus] = useState<string>("ALL");
   207→  const [busyId, setBusyId] = useState<string | null>(null);
   208→  const [editing, setEditing] = useState<TenantVehicleRowDTO | null>(null);
   209→  const [editPrice, setEditPrice] = useState("");
   210→
   211→  const url = `/api/tenant/inventory${qs({ status: status === "ALL" ? undefined : status })}`;
   212→  const { data, loading, error, refetch } = useFetch<TenantVehicleRowDTO[]>(url);
   213→
   214→  const patch = async (id: string, body: Record<string, string | number | boolean>) => {
   215→    setBusyId(id);
   216→    try {
   217→      await apiSend(`/api/tenant/inventory/${id}`, "PATCH", body);
   218→      toast.success(t("toastUpdated"));
   219→      refetch();
   220→    } catch (e) {
   221→      toast.error(isForbidden(e) ? t("toastForbidden") : isFeaturedLimit(e) ? t("toastFeaturedLimit") : errorText(e));
   222→    } finally {
   223→      setBusyId(null);
   224→    }
   225→  };
   226→
   227→  const submitPrice = async () => {
   228→    if (!editing) return;
   229→    const p = Number(editPrice);
   230→    if (!Number.isFinite(p) || p <= 0) return;
   231→    await patch(editing.id, { priceUsd: p });
   232→    setEditing(null);
   233→  };
   234→
   235→  return (
   236→    <div className="space-y-4">
   237→      <div className="flex flex-wrap items-center justify-between gap-2">
   238→        <div className="flex flex-wrap gap-1.5">
   239→          {TENANT_STATUSES.map((s) => (
   240→            <button
   241→              key={s}
   242→              onClick={() => setStatus(s)}
   243→              aria-pressed={status === s}
   244→              className={cn(
   245→                "rounded-full border px-3 py-1.5 text-xs font-medium transition-colors cursor-pointer",
   246→                status === s ? "border-amber-500 bg-amber-50 text-amber-800" : "border-slate-200 bg-white text-slate-600 hover:border-slate-300"
   247→              )}
   248→            >
   249→              {t(`vst${s}` as never)}
   250→            </button>
   251→          ))}
   252→        </div>
   253→        <Button size="sm" onClick={() => navigate({ name: "sell" })} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   254→          <Plus className="h-4 w-4 mr-1.5" aria-hidden />
   255→          {t("addListing")}
   256→        </Button>
   257→      </div>
   258→
   259→      {loading && !data ? (
   260→        <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>
   261→      ) : error && !data ? (
   262→        <FetchErrorState message={error} onRetry={refetch} />
   263→      ) : !data || data.length === 0 ? (
   264→        <EmptyState icon={Car} text={t("inventoryEmpty")} />
   265→      ) : (
   266→        <div className="space-y-2.5">
   267→          {data.map((v) => (
   268→            <div key={v.id} className="flex items-center gap-3 rounded-xl border border-slate-200 bg-white p-3">
   269→              <div className="h-14 w-20 shrink-0 overflow-hidden rounded-lg bg-slate-100">
   270→                {v.images[0] ? (
   271→                  <img src={v.images[0]} alt="" className="h-full w-full object-cover" />
   272→                ) : (
   273→                  <Car className="mx-auto h-6 w-6 text-slate-300" aria-hidden />
   274→                )}
   275→              </div>
   276→              <div className="min-w-0 flex-1">
   277→                <p className="truncate text-sm font-medium text-slate-900">
   278→                  {v.make} {v.model} {v.variant ?? ""} {v.year}
   279→                </p>
   280→                <p className="mt-0.5 text-[11px] text-slate-400">
   281→                  {formatUsd(v.priceUsd)} · <Eye className="inline h-3 w-3" aria-hidden /> {v.views}
   282→                </p>
   283→                <div className="mt-1.5 flex items-center gap-2">
   284→                  <VehicleStatusBadge status={v.status} />
   285→                  {v.featured && <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" aria-label={t("actFeature")} />}
   286→                </div>
   287→              </div>
   288→              <DropdownMenu>
   289→                <DropdownMenuTrigger asChild>
   290→                  <Button variant="outline" size="icon" className="h-9 w-9" disabled={busyId === v.id} aria-label={`${t("actions")} ${v.make} ${v.model}`}>
   291→                    {busyId === v.id ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <MoreHorizontal className="h-4 w-4" aria-hidden />}
   292→                  </Button>
   293→                </DropdownMenuTrigger>
   294→                <DropdownMenuContent align="end" className="w-48">
   295→                  <DropdownMenuLabel>{t("actions")}</DropdownMenuLabel>
   296→                  <DropdownMenuSeparator />
   297→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "PUBLISHED" })}>{t("actPublish")}</DropdownMenuItem>
   298→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "DRAFT" })}>{t("actUnpublish")}</DropdownMenuItem>
   299→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "SOLD" })}>{t("actMarkSold")}</DropdownMenuItem>
   300→                  <DropdownMenuItem onClick={() => patch(v.id, { status: "ARCHIVED" })}>{t("actArchive")}</DropdownMenuItem>
   301→                  <DropdownMenuSeparator />
   302→                  <DropdownMenuItem onClick={() => patch(v.id, { featured: !v.featured })}>
   303→                    {v.featured ? t("actUnfeature") : t("actFeature")}
   304→                  </DropdownMenuItem>
   305→                  <DropdownMenuItem onClick={() => { setEditing(v); setEditPrice(String(v.priceUsd)); }}>
   306→                    {t("actEditPrice")}
   307→                  </DropdownMenuItem>
   308→                </DropdownMenuContent>
   309→              </DropdownMenu>
   310→            </div>
   311→          ))}
   312→        </div>
   313→      )}
   314→
   315→      {/* Edit price dialog */}
   316→      <Dialog open={Boolean(editing)} onOpenChange={(o) => !o && setEditing(null)}>
   317→        <DialogContent className="sm:max-w-xs">
   318→          <DialogHeader>
   319→            <DialogTitle>{t("editPriceTitle")}</DialogTitle>
   320→          </DialogHeader>
   321→          <div>
   322→            <Label htmlFor="inv-price" className="text-xs text-slate-500">{editing ? `${editing.make} ${editing.model} ${editing.year}` : ""}</Label>
   323→            <Input
   324→              id="inv-price" type="number" min={0} className="mt-1.5 h-9" value={editPrice}
   325→              onChange={(e) => setEditPrice(e.target.value)} autoFocus
   326→            />
   327→          </div>
   328→          <DialogFooter>
   329→            <Button variant="outline" size="sm" onClick={() => setEditing(null)}>{t("cancel")}</Button>
   330→            <Button size="sm" onClick={submitPrice} disabled={busyId !== null} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   331→              {t("saveChanges")}
   332→            </Button>
   333→          </DialogFooter>
   334→        </DialogContent>
   335→      </Dialog>
   336→    </div>
   337→  );
   338→}

   403→function DealerAppointments() {
   404→  const t = useT();
   405→  const locale = useAppStore((s) => s.locale);
   406→  const { data, loading, error, refetch } = useFetch<TenantAppointmentDTO[]>("/api/tenant/appointments");
   407→  const [busyId, setBusyId] = useState<string | null>(null);
   408→
   409→  const patch = async (id: string, status: string) => {
   410→    setBusyId(id);
   411→    try {
   412→      await apiSend(`/api/tenant/appointments/${id}`, "PATCH", { status });
   413→      toast.success(t("toastUpdated"));
   414→      refetch();
   415→    } catch (e) {
   416→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   417→    } finally {
   418→      setBusyId(null);
   419→    }
   420→  };
   421→
   422→  if (loading && !data) return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>;
   423→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   424→  if (!data || data.length === 0) return <EmptyState icon={CalendarDays} text={t("apptEmpty")} hint={t("apptEmptyHint")} />;
   425→

   620→  const [requesting, setRequesting] = useState<PlanDTO | null>(null);
   621→  const [downgrading, setDowngrading] = useState(false);
   622→  const [historyKey, setHistoryKey] = useState(0);
   623→

   625→    if (data?.subscription) setCycle(data.subscription.billingCycle);
   626→  }, [data?.subscription?.billingCycle]);

   630→    try {
   631→      await apiSend("/api/tenant/subscription", "POST", { plan: "FREE" });
   632→      toast.success(t("toastUpdated"));
   633→      refetch();
   634→    } catch (e) {
   635→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));

   637→      setDowngrading(false);
   638→    }

   642→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   643→  const plans = data?.plans ?? [];
   644→  const current = data?.subscription ?? null;
   645→  const expired = current?.status === "EXPIRED";
   646→  const freeDisabled = data?.freePlanEnabled === false;
   647→  const paymentsDisabled = data?.paymentsEnabled === false;
   648→
   649→  return (
   650→    <div className="space-y-4">
   651→      {paymentsDisabled && (
   652→        <Alert className="border-amber-300 bg-amber-50">
   653→          <AlertDescription className="text-xs font-medium text-amber-900">{t("payDisabledNotice")}</AlertDescription>
   654→        </Alert>

   660→      <div className="flex flex-wrap items-center justify-between gap-2">

   663→          {(["MONTHLY", "YEARLY"] as const).map((c) => (
   664→            <button
   665→              key={c}
   666→              onClick={() => setCycle(c)}
   667→              aria-pressed={cycle === c}
   668→              className={cn(
   669→                "rounded-full px-3 py-1.5 text-xs font-medium transition-colors cursor-pointer",
   670→                cycle === c ? "bg-amber-500 text-slate-950" : "text-slate-600 hover:text-slate-900"
   671→              )}
   672→            >
   673→              {c === "MONTHLY" ? t("billingMonthly") : t("billingYearly")}
   674→            </button>
   675→          ))}
   676→        </div>
   677→      </div>
   678→
   679→      {current && (
   680→        <div className={cn("rounded-xl border p-4 text-sm", expired ? "border-rose-200 bg-rose-50/60" : "border-amber-200 bg-amber-50/60")}>
   681→          <p className={cn("font-semibold", expired ? "text-rose-900" : "text-amber-900")}>
   682→            {current.plan} · {formatUsd(current.priceUsd)} / {current.billingCycle === "MONTHLY" ? t("billingMonthly").toLowerCase() : t("billingYearly").toLowerCase()}
   683→            {expired && <span className="ml-2 rounded-full bg-rose-100 px-2 py-0.5 text-[10px] font-bold text-rose-800">{t("subExpiresBadge")}</span>}

   686→            {t("subRenews", { d: formatDateTime(current.currentPeriodEnd, locale) })} · {t("subSeats", { n: current.seats })}
   687→          </p>
   688→          {(expired || current.status === "PAST_DUE") && (
   689→            <p className="mt-1.5 text-xs font-medium text-rose-800">{t("subPayFirst")}</p>
   690→          )}
   691→        </div>
   692→      )}
   693→
   694→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   695→        {plans.map((p) => {
   696→          const isCurrent = current?.plan === p.plan && !expired;
   697→          const price = cycle === "MONTHLY" ? p.priceUsdMonthly : p.priceUsdYearly;
   698→          const isPaid = p.plan !== "FREE";
   699→          const freeBlocked = p.plan === "FREE" && freeDisabled;
   700→          return (
   701→            <div
   702→              key={p.plan}
   703→              className={cn(
   704→                "flex flex-col rounded-xl border bg-white p-4",
   705→                isCurrent ? "border-amber-400 ring-1 ring-amber-200" : "border-slate-200",
   706→                freeBlocked && "opacity-70"
   707→              )}
   708→            >
   709→              <div className="flex items-center justify-between">
   710→                <p className="text-sm font-bold text-slate-900">{p.plan}</p>
   711→                {isCurrent && <span className="rounded-full bg-amber-100 px-2 py-0.5 text-[10px] font-bold text-amber-800">{t("subCurrentBadge")}</span>}
   712→              </div>
   713→              <p className="mt-2 text-2xl font-bold tracking-tight text-slate-900">
   714→                {p.plan === "FREE" ? "$0" : formatUsd(price)}
   715→                {p.plan !== "FREE" && <span className="text-xs font-medium text-slate-400"> / {cycle === "MONTHLY" ? t("billingMonthly").toLowerCase() : t("billingYearly").toLowerCase()}</span>}
   716→              </p>
   717→              <ul className="mt-3 space-y-1 text-xs text-slate-500">
   718→                <li>· {t("kpiListings")}: {p.listingsLimit}</li>
   719→                <li>· {t("actFeature")}: {p.featuredLimit}</li>
   720→                <li>· {t("subSeats", { n: p.seats })}</li>
   721→                {p.features.map((f) => (
   722→                  <li key={f.en}>· {f[locale]}</li>
   723→                ))}
   724→              </ul>
   725→              {freeBlocked && (
   726→                <p className="mt-2 text-[11px] font-medium text-rose-700">{t("payFreeDisabledNotice")}</p>
   727→              )}
   728→              <Button
   729→                size="sm"
   730→                className="mt-4 w-full"
   731→                variant={isCurrent ? "secondary" : "default"}
   732→                disabled={isCurrent || downgrading || freeBlocked || (p.plan === "FREE" && (!current || current.plan === "FREE"))}
   733→                onClick={() => (isPaid ? setRequesting(p) : downgrade())}
   734→                aria-label={isPaid ? t("payOpen") : t("subFreeDowngrade")}
   735→              >
   736→                {downgrading && !isPaid && <Loader2 className="h-4 w-4 mr-1.5 animate-spin" aria-hidden />}
   737→                {isCurrent ? t("subCurrentBadge") : isPaid ? (current && current.plan !== "FREE" ? t("payRenew") : t("payOpen")) : t("subFreeDowngrade")}
   738→              </Button>
   739→            </div>
   740→          );
   741→        })}
   742→      </div>
   743→
   744→      <TenantPaymentHistory key={historyKey} onChanged={refetch} />
   745→
   746→      <RequestSubscriptionDialog
   747→        open={requesting !== null}
   748→        onOpenChange={(o) => !o && setRequesting(null)}
   749→        plan={requesting}
   750→        cycle={cycle}
   751→        priceUsd={requesting ? (cycle === "MONTHLY" ? requesting.priceUsdMonthly : requesting.priceUsdYearly) : 0}
   752→        onDone={() => { refetch(); setRequesting(null); setHistoryKey((k) => k + 1); }}
   753→      />
   754→    </div>
   755→  );
   756→}

   100→function AdminOverview() {

   206→function AdminTenants() {
   207→  const t = useT();
   208→  const { data, loading, error, refetch } = useFetch<AdminTenantDTO[]>("/api/admin/tenants");
   209→  const [busyId, setBusyId] = useState<string | null>(null);
   210→  const [extending, setExtending] = useState<{ id: string; name: string } | null>(null);
   211→  const [extendMonths, setExtendMonths] = useState("1");
   212→  const [extendNote, setExtendNote] = useState("");
   213→  const [extendingBusy, setExtendingBusy] = useState(false);
   214→
   215→  const submitExtend = async () => {
   216→    if (!extending) return;
   217→    const months = Number(extendMonths);
   218→    if (!Number.isInteger(months) || months < 1 || months > 36) return;
   219→    setExtendingBusy(true);
   220→    try {
   221→      await apiSend("/api/admin/subscriptions/extend", "POST", {
   222→        tenantId: extending.id,
   223→        months,
   224→        note: extendNote.trim() || undefined,
   225→      });
   226→      toast.success(t("subExtendedToast"));
   227→      setExtending(null);
   228→      setExtendMonths("1");
   229→      setExtendNote("");
   230→      refetch();
   231→    } catch (e) {
   232→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   233→    } finally {
   234→      setExtendingBusy(false);
   235→    }
   236→  };
   237→
   238→  const patch = async (id: string, body: Record<string, string>) => {
   239→    setBusyId(id);
   240→    try {
   241→      await apiSend(`/api/admin/tenants/${id}`, "PATCH", body);
   242→      toast.success(t("toastUpdated"));
   243→      refetch();
   244→    } catch (e) {
   245→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   246→    } finally {
   247→      setBusyId(null);
   248→    }
   249→  };
   250→
   251→  if (loading && !data) {
   252→    return <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}</div>;
   253→  }
   254→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   255→  if (!data || data.length === 0) return <EmptyState icon={Store} text={t("noResults")} />;
   256→
   257→  return (
   258→    <div className="space-y-3">
   259→      {data.map((tt) => (
   260→        <div key={tt.id} className="rounded-xl border border-slate-200 bg-white p-4">
   261→          <div className="flex flex-wrap items-start justify-between gap-3">
   262→            <div className="min-w-0">
   263→              <p className="flex items-center gap-2 text-sm font-semibold text-slate-900">
   264→                <span className="truncate">{tt.name}</span>
   265→                <VerificationBadge verification={tt.verification} />
   266→                <TenantStatusBadge status={tt.status} />
   267→              </p>
   268→              <p className="mt-0.5 text-[11px] text-slate-400">
   269→                {tt.slug} · {tt.city} · {t("sellerSince", { y: new Date(tt.createdAt).getFullYear() })}
   270→              </p>
   271→            </div>
   272→            <DropdownMenu>
   273→              <DropdownMenuTrigger asChild>
   274→                <Button variant="outline" size="sm" disabled={busyId === tt.id} aria-label={`${t("actions")} ${tt.name}`}>
   275→                  {busyId === tt.id ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <MoreHorizontal className="h-4 w-4" aria-hidden />}
   276→                  <ChevronDown className="h-3 w-3 opacity-60" aria-hidden />
   277→                </Button>
   278→              </DropdownMenuTrigger>
   279→              <DropdownMenuContent align="end" className="w-44">
   280→                <DropdownMenuLabel>{t("actions")}</DropdownMenuLabel>
   281→                <DropdownMenuSeparator />
   282→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "VERIFIED" })}>{t("actVerify")}</DropdownMenuItem>
   283→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "PENDING" })}>{t("actSetPending")}</DropdownMenuItem>
   284→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "REJECTED" })} className="text-rose-600 focus:text-rose-700">
   285→                  {t("actReject")}
   286→                </DropdownMenuItem>
   287→                <DropdownMenuSeparator />
   288→                <DropdownMenuItem onClick={() => patch(tt.id, { status: "SUSPENDED" })} className="text-rose-600 focus:text-rose-700">
   289→                  {t("actSuspend")}
   290→                </DropdownMenuItem>
   291→                <DropdownMenuItem onClick={() => patch(tt.id, { status: "ACTIVE" })}>{t("actActivate")}</DropdownMenuItem>
   292→                <DropdownMenuSeparator />
   293→                <DropdownMenuItem onClick={() => setExtending({ id: tt.id, name: tt.name })}>{t("subManualExtend")}</DropdownMenuItem>
   294→              </DropdownMenuContent>
   295→            </DropdownMenu>
   296→          </div>
   297→          <div className="mt-3 grid grid-cols-2 gap-2 text-xs sm:grid-cols-4">
   298→            <div className="rounded-lg bg-slate-50 px-3 py-2">
   299→              <p className="text-[10px] text-slate-400">{t("tabStaff")}</p>
   300→              <p className="font-semibold text-slate-700">{tt.memberCount}</p>
   301→            </div>
   302→            <div className="rounded-lg bg-slate-50 px-3 py-2">
   303→              <p className="text-[10px] text-slate-400">{t("kpiListings")}</p>
   304→              <p className="font-semibold text-slate-700">{tt.listingCount}</p>
   305→            </div>
   306→            <div className="rounded-lg bg-slate-50 px-3 py-2">
   307→              <p className="text-[10px] text-slate-400">{t("tabSubscription")}</p>
   308→              <p className="font-semibold text-slate-700">{tt.plan ?? "—"}</p>
   309→            </div>
   310→            <div className="rounded-lg bg-slate-50 px-3 py-2">
   311→              <p className="text-[10px] text-slate-400">{t("kpiMrr")}</p>
   312→              <p className="font-semibold text-slate-700">{formatUsd(tt.mrrUsd)}</p>
   313→            </div>
   314→          </div>
   315→        </div>
   316→      ))}
   317→
   318→      {/* Manual subscription extension dialog */}
   319→      <Dialog open={Boolean(extending)} onOpenChange={(o) => !o && setExtending(null)}>
   320→        <DialogContent className="sm:max-w-sm">
   321→          <DialogHeader>
   322→            <DialogTitle>{t("subManualExtend")}</DialogTitle>
   323→          </DialogHeader>
   324→          <div className="space-y-3">
   325→            <p className="text-sm text-slate-500">{extending?.name}</p>
   326→            <div>
   327→              <Label htmlFor="ext-months" className="text-xs text-slate-500">{t("subExtendMonths")}</Label>
   328→              <Input id="ext-months" type="number" min={1} max={36} value={extendMonths}
   329→                onChange={(e) => setExtendMonths(e.target.value)} className="mt-1.5 h-9" />
   330→            </div>
   331→            <div>
   332→              <Label htmlFor="ext-note" className="text-xs text-slate-500">{t("subExtendNote")}</Label>
   333→              <Textarea id="ext-note" value={extendNote} onChange={(e) => setExtendNote(e.target.value)} rows={2} className="mt-1.5" maxLength={300} />
   334→            </div>
   335→          </div>
   336→          <DialogFooter>
   337→            <Button variant="outline" size="sm" onClick={() => setExtending(null)}>{t("cancel")}</Button>
   338→            <Button size="sm" onClick={submitExtend} disabled={extendingBusy} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   339→              {extendingBusy && <Loader2 className="h-4 w-4 mr-2 animate-spin" aria-hidden />}
   340→              {t("subManualExtend")}
   341→            </Button>
   342→          </DialogFooter>
   343→        </DialogContent>
   344→      </Dialog>
   345→    </div>
   346→  );
   347→}

   349→function AdminListings() {
   350→  const t = useT();
   351→  const navigate = useAppStore((s) => s.navigate);
   352→  const [status, setStatus] = useState<string>("PENDING_REVIEW");
   353→  const [q, setQ] = useState("");
   354→  const [debQ, setDebQ] = useState("");
   355→  const [busyId, setBusyId] = useState<string | null>(null);
   356→
   357→  useEffect(() => {
   358→    const h = setTimeout(() => setDebQ(q.trim()), 350);
   359→    return () => clearTimeout(h);
   360→  }, [q]);
   361→
   362→  const url = `/api/admin/vehicles${qs({ status: status === "ALL" ? undefined : status, q: debQ || undefined })}`;
   363→  const { data, loading, error, refetch } = useFetch<AdminVehicleRowDTO[]>(url);
   364→
   365→  const patch = async (id: string, body: Record<string, string | boolean>) => {
   366→    setBusyId(id);
   367→    try {
   368→      await apiSend(`/api/admin/vehicles/${id}`, "PATCH", body);
   369→      toast.success(t("toastUpdated"));
   370→      refetch();
   371→    } catch (e) {
   372→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   373→    } finally {
   374→      setBusyId(null);
   375→    }
   376→  };
   377→
   378→  const chips = ["ALL", ...VEHICLE_STATUSES];
   379→
   380→  return (
   381→    <div className="space-y-4">
   382→      <div className="flex flex-wrap items-center gap-2">
   383→        <div className="flex w-full items-center gap-2 sm:w-64">
   384→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder={`${t("searchButton")}…`} className="h-9" aria-label={t("searchButton")} />
   385→        </div>
   386→        <div className="flex flex-wrap gap-1.5">
   387→          {chips.map((c) => (
   388→            <button
   389→              key={c}
   390→              onClick={() => setStatus(c)}
   391→              aria-pressed={status === c}
   392→              className={cn(
   393→                "rounded-full border px-3 py-1.5 text-xs font-medium transition-colors cursor-pointer min-h-[32px]",
   394→                status === c ? "border-amber-500 bg-amber-50 text-amber-800" : "border-slate-200 bg-white text-slate-600 hover:border-slate-300"
   395→              )}
   396→            >
   397→              {t(`vst${c}` as never)}
   398→            </button>
   399→          ))}
   400→        </div>
   401→      </div>
   402→
   403→      {loading && !data ? (
   404→        <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>
   405→      ) : error && !data ? (
   406→        <FetchErrorState message={error} onRetry={refetch} />
   407→      ) : !data || data.length === 0 ? (
   408→        <EmptyState icon={Car} text={t("noResults")} />
   409→      ) : (
   410→        <div className="space-y-2.5">
   411→          {data.map((v) => (
   412→            <div key={v.id} className="flex items-center gap-3 rounded-xl border border-slate-200 bg-white p-3">
   413→              <button
   414→                onClick={() => navigate({ name: "detail", vehicleId: v.id })}
   415→                className="h-14 w-20 shrink-0 overflow-hidden rounded-lg bg-slate-100 cursor-pointer"
   416→                aria-label={`${v.make} ${v.model} ${v.year}`}
   417→              >
   418→                {v.images[0] ? (
   419→                  <img src={v.images[0]} alt="" className="h-full w-full object-cover" />
   420→                ) : (
   421→                  <Car className="mx-auto h-6 w-6 text-slate-300" aria-hidden />
   422→                )}
   423→              </button>
   424→              <div className="min-w-0 flex-1">
   425→                <p className="truncate text-sm font-medium text-slate-900">
   426→                  {v.make} {v.model} {v.variant ?? ""} {v.year}
   427→                </p>
   428→                <p className="mt-0.5 truncate text-[11px] text-slate-400">
   429→                  {formatUsd(v.priceUsd)} · {v.sellerName} · {v.city}
   430→                </p>
   431→                <div className="mt-1.5 flex items-center gap-2">
   432→                  <VehicleStatusBadge status={v.status} />
   433→                  {v.featured && (
   434→                    <span className="inline-flex items-center gap-1 text-[10px] font-semibold text-amber-600">
   435→                      <Star className="h-3 w-3 fill-amber-400 text-amber-400" aria-hidden />
   436→                    </span>
   437→                  )}
   438→                </div>
   439→              </div>
   440→              <div className="flex shrink-0 items-center gap-1.5">
   441→                <button
   442→                  onClick={() => patch(v.id, { featured: !v.featured })}
   443→                  disabled={busyId === v.id}
   444→                  className={cn(
   445→                    "flex h-9 w-9 items-center justify-center rounded-lg border transition-colors cursor-pointer disabled:opacity-50",
   446→                    v.featured ? "border-amber-300 bg-amber-50 text-amber-500" : "border-slate-200 text-slate-400 hover:text-amber-500"
   447→                  )}
   448→                  aria-label={v.featured ? t("actUnfeature") : t("actFeature")}
   449→                >

   765→function AdminPayments() {

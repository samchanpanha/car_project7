    30→/** Local "YYYY-MM-DDTHH:mm" for datetime-local min attribute. */
    31→function localInputValue(d: Date): string {
    32→  const pad = (n: number) => String(n).padStart(2, "0");
    33→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
    34→}

   427→                  <span className="text-2xl font-bold text-teal-700">{Math.round(v.report.overallConfidence)}%</span>
   428→                  <span className="text-xs text-slate-500">{t("confidence")}</span>

   431→                  onClick={() => navigate({ name: "report", vehicleId: v.id })}

   433→                >

   436→                </Button>
   437→              </div>
   438→            ) : (
   439→              <Button
   440→                onClick={generateReport}
   441→                disabled={generating}
   442→                className="mt-3 w-full bg-teal-600 hover:bg-teal-700 text-white font-semibold"
   443→              >
   444→                {generating ? (
   445→                  <>
   446→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" aria-hidden />

   448→                  </>
   449→                ) : (
   450→                  <>
   451→                    <FileCheck2 className="h-4 w-4 mr-2" aria-hidden />

   453→                  </>
   454→                )}

   458→
   459→          {/* Safety */}

   469→

   481→

   484→        open={contactOpen}

   487→        sellerName={sellerName}

   489→

   509→          </DialogHeader>

   530→            <Button
   531→              size="sm"
   532→              onClick={submitTestDrive}
   533→              disabled={tdSubmitting || !tdWhen}
   534→              className="bg-teal-600 hover:bg-teal-700 text-white font-semibold"

   542→
   543→      {/* Sticky mobile CTA bar */}
   544→      <div className="fixed bottom-[52px] inset-x-0 z-30 border-t border-slate-200 bg-white/95 backdrop-blur p-2.5 md:hidden">
   545→        <div className="flex gap-2 max-w-md mx-auto">
   546→          <Button
   547→            onClick={() => setContactOpen(true)}
   548→            className="flex-1 bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold h-11"
   549→          >
   550→            <Send className="h-4 w-4 mr-2" aria-hidden />
   551→            {t("contactSeller")}
   552→          </Button>
   553→          <Button onClick={toggleFavorite} variant="outline" size="icon" className="h-11 w-11" aria-label={t("navFavorites")}>
   554→            <Heart className={cn("h-5 w-5", isFav && "fill-rose-500 text-rose-500")} aria-hidden />
   555→          </Button>
   556→          {v.report && (
   557→            <Button
   558→              onClick={() => navigate({ name: "report", vehicleId: v.id })}
   559→              variant="outline"
   560→              size="icon"
   561→              className="h-11 w-11 border-teal-300 text-teal-700"

   563→            >
   564→              <RefreshCw className="hidden" aria-hidden />
   565→              <FileCheck2 className="h-5 w-5" aria-hidden />
   566→            </Button>
   567→          )}
   568→        </div>
   569→      </div>
   570→    </div>
   571→  );
   572→}

   574→function ReportDialog({
   575→  open,
   576→  onOpenChange,
   577→  reason,
   578→  setReason,
   579→  details,
   580→  setDetails,
   581→  onSubmit,
   582→}: {
   583→  open: boolean;
   584→  onOpenChange: (v: boolean) => void;
   585→  reason: (typeof ABUSE_REASONS)[number];
   586→  setReason: (r: (typeof ABUSE_REASONS)[number]) => void;
   587→  details: string;
   588→  setDetails: (d: string) => void;
   589→  onSubmit: () => void;
   590→}) {
   591→  const t = useT();
   592→  const reasonLabels: Record<string, string> = {
   593→    SCAM: "Scam / បញ្ឆោត",
   594→    WRONG_INFO: "Wrong information / ព័ត៌មានខុស",
   595→    DUPLICATE: "Duplicate / ស្ទួន",
   596→    OFFENSIVE: "Offensive / មិនសមរម្យ",
   597→    SOLD: "Already sold / លក់រួច",
   598→    OTHER: "Other / ផ្សេងៗ",
   599→  };
   600→  return (
   601→    <ReportDialogWrapper open={open} onOpenChange={onOpenChange} onSubmit={onSubmit}>
   602→      <p className="text-sm font-medium text-slate-700 mb-2">{t("reportReason")}</p>
   603→      <div className="grid grid-cols-2 gap-1.5">
   604→        {ABUSE_REASONS.map((r) => (
   605→          <button
   606→            key={r}
   607→            onClick={() => setReason(r)}
   608→            className={cn(
   609→              "rounded-lg border px-2.5 py-1.5 text-xs font-medium transition-colors cursor-pointer",
   610→              reason === r
   611→                ? "border-rose-400 bg-rose-50 text-rose-700"
   612→                : "border-slate-200 text-slate-600 hover:border-slate-300"
   613→            )}
   614→            aria-pressed={reason === r}
   615→          >
   616→            {reasonLabels[r]}
   617→          </button>
   618→        ))}
   619→      </div>
   620→      <textarea
   621→        value={details}
   622→        onChange={(e) => setDetails(e.target.value)}
   623→        rows={3}
   624→        placeholder={t("reportDetails")}
   625→        className="mt-3 w-full rounded-lg border border-slate-200 p-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-amber-500"
   626→      />
   627→    </ReportDialogWrapper>
   628→  );
   629→}

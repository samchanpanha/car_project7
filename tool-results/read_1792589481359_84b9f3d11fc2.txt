    52→    52→    52→/** Read a file as a raw data URL — used for PDFs (no canvas downscale). */
    53→    53→    53→function rawFileToDataUrl(file: File): Promise<string> {
    54→    54→    54→  return new Promise((resolve, reject) => {
    55→    55→    55→    const reader = new FileReader();
    56→    56→    56→    reader.onload = () => resolve(String(reader.result));
    57→    57→    57→    reader.onerror = () => reject(new Error("read-failed"));
    58→    58→    58→    reader.readAsDataURL(file);
    59→    59→    59→  });
    60→    60→    60→}

   286→   286→   286→/**
   287→   287→   287→ * Unlimited photo manager: multi-file upload (client-downscaled, stored via
   288→   288→   288→ * POST /api/uploads), link adds, remove, set-as-cover. `images` is the ordered
   289→   289→   289→ * desired list — first entry is the cover.
   290→   290→   290→ */
   291→   291→   291→export function PhotoUploader({
   292→   292→   292→  images,
   293→   293→   293→  onChange,
   294→   294→   294→  disabled,
   295→   295→   295→}: {
   296→   296→   296→  images: string[];
   297→   297→   297→  onChange: (next: string[]) => void;
   298→   298→   298→  disabled?: boolean;
   299→   299→   299→}) {
   300→   300→   300→  const t = useT();
   301→   301→   301→  const inputRef = useRef<HTMLInputElement>(null);
   302→   302→   302→  const [uploading, setUploading] = useState(0);
   303→   303→   303→  const [urlInput, setUrlInput] = useState("");
   304→   304→   304→  // Latest list ref so in-flight uploads never resurrect a just-removed photo.
   305→   305→   305→  const latest = useRef(images);
   306→   306→   306→  latest.current = images;
   307→   307→   307→
   308→   308→   308→  const commit = (next: string[]) => onChange(Array.from(new Set(next)));
   309→   309→   309→
   310→   310→   310→  const uploadFiles = async (files: File[]) => {
   311→   311→   311→    const list = files.filter((f) => f.type.startsWith("image/"));
   312→   312→   312→    if (list.length === 0) return;
   313→   313→   313→    setUploading(list.length);
   314→   314→   314→    let failed = 0;
   315→   315→   315→    const added: string[] = [];
   316→   316→   316→    for (const file of list) {
   317→   317→   317→      try {
   318→   318→   318→        const dataUrl = await fileToDataUrl(file, 1280, 0.82);
   319→   319→   319→        const res = await apiSend<{ url: string }>("/api/uploads", "POST", { dataUrl });
   320→   320→   320→        added.push(res.url);
   321→   321→   321→      } catch {
   322→   322→   322→        failed += 1;
   323→   323→   323→      } finally {
   324→   324→   324→        setUploading((n) => Math.max(0, n - 1));
   325→   325→   325→      }
   326→   326→   326→    }
   327→   327→   327→    if (added.length > 0) commit([...latest.current, ...added]);
   328→   328→   328→    if (failed > 0) toast.error(t("uploadFailedCount", { n: failed }));
   329→   329→   329→  };
   330→   330→   330→
   331→   331→   331→  const addUrl = () => {
   332→   332→   332→    const u = urlInput.trim();
   333→   333→   333→    if (!u || !/^(https?:\/\/|\/api\/uploads\/)/.test(u)) {
   334→   334→   334→      toast.warning(t("invalidPhotoUrl"));
   335→   335→   335→      return;
   336→   336→   336→    }
   337→   337→   337→    if (!latest.current.includes(u)) commit([...latest.current, u]);
   338→   338→   338→    setUrlInput("");
   339→   339→   339→  };
   340→   340→   340→
   341→   341→   341→  const removeAt = (idx: number) => commit(images.filter((_, i) => i !== idx));
   342→   342→   342→  const makeCover = (idx: number) => {
   343→   343→   343→    if (idx === 0) return;
   344→   344→   344→    commit([images[idx], ...images.filter((_, i) => i !== idx)]);
   345→   345→   345→  };
   346→   346→   346→
   347→   347→   347→  return (
   348→   348→   348→    <div>
   349→   349→   349→      <div className="flex flex-wrap items-center gap-2">
   350→   350→   350→        <input
   351→   351→   351→          ref={inputRef}
   352→   352→   352→          type="file"
   353→   353→   353→          accept="image/*"
   354→   354→   354→          multiple
   355→   355→   355→          className="hidden"
   356→   356→   356→          aria-label={t("uploadPhotos")}
   357→   357→   357→          onChange={(e) => {
   358→   358→   358→            if (e.target.files) void uploadFiles(Array.from(e.target.files));
   359→   359→   359→            e.target.value = "";
   360→   360→   360→          }}
   361→   361→   361→        />
   362→   362→   362→        <Button
   363→   363→   363→          type="button"
   364→   364→   364→          variant="outline"
   365→   365→   365→          size="sm"
   366→   366→   366→          className="h-9 border-amber-300 text-amber-800 hover:bg-amber-50 hover:text-amber-900"
   367→   367→   367→          disabled={disabled || uploading > 0}
   368→   368→   368→          onClick={() => inputRef.current?.click()}
   369→   369→   369→        >
   370→   370→   370→          {uploading > 0 ? (
   371→   371→   371→            <>
   372→   372→   372→              <Loader2 className="h-4 w-4 mr-1.5 animate-spin" aria-hidden />
   373→   373→   373→              {t("uploadingPhotos", { n: uploading })}
   374→   374→   374→            </>
   375→   375→   375→          ) : (
   376→   376→   376→            <>
   377→   377→   377→              <ImagePlus className="h-4 w-4 mr-1.5" aria-hidden />
   378→   378→   378→              {t("uploadPhotos")}
   379→   379→   379→            </>
   380→   380→   380→          )}
   381→   381→   381→        </Button>
   382→   382→   382→        <span className="text-[11px] text-slate-400">{t("photosUnlimitedHint")}</span>
   383→   383→   383→      </div>
   384→   384→   384→

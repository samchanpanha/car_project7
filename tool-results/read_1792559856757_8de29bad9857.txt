   199→   199→function AdminTenants() {
   200→   200→  const t = useT();
   201→   201→  const { data, loading, error, refetch } = useFetch<AdminTenantDTO[]>("/api/admin/tenants");
   202→   202→  const [busyId, setBusyId] = useState<string | null>(null);
   203→   203→
   204→   204→  const patch = async (id: string, body: Record<string, string>) => {
   205→   205→    setBusyId(id);
   206→   206→    try {
   207→   207→      await apiSend(`/api/admin/tenants/${id}`, "PATCH", body);
   208→   208→      toast.success(t("toastUpdated"));
   209→   209→      refetch();
   210→   210→    } catch (e) {
   211→   211→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   212→   212→    } finally {
   213→   213→      setBusyId(null);
   214→   214→    }
   215→   215→  };
   216→   216→
   217→   217→  if (loading && !data) {
   218→   218→    return <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}</div>;
   219→   219→  }
   220→   220→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   221→   221→  if (!data || data.length === 0) return <EmptyState icon={Store} text={t("noResults")} />;
   222→   222→
   223→   223→  return (
   224→   224→    <div className="space-y-3">
   225→   225→      {data.map((tt) => (
   226→   226→        <div key={tt.id} className="rounded-xl border border-slate-200 bg-white p-4">
   227→   227→          <div className="flex flex-wrap items-start justify-between gap-3">
   228→   228→            <div className="min-w-0">
   229→   229→              <p className="flex items-center gap-2 text-sm font-semibold text-slate-900">
   230→   230→                <span className="truncate">{tt.name}</span>
   231→   231→                <VerificationBadge verification={tt.verification} />
   232→   232→                <TenantStatusBadge status={tt.status} />
   233→   233→              </p>
   234→   234→              <p className="mt-0.5 text-[11px] text-slate-400">
   235→   235→                {tt.slug} · {tt.city} · {t("sellerSince", { y: new Date(tt.createdAt).getFullYear() })}
   236→   236→              </p>
   237→   237→            </div>
   238→   238→            <DropdownMenu>
   239→   239→              <DropdownMenuTrigger asChild>
   240→   240→                <Button variant="outline" size="sm" disabled={busyId === tt.id} aria-label={`${t("actions")} ${tt.name}`}>
   241→   241→                  {busyId === tt.id ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <MoreHorizontal className="h-4 w-4" aria-hidden />}
   242→   242→                  <ChevronDown className="h-3 w-3 opacity-60" aria-hidden />
   243→   243→                </Button>
   244→   244→              </DropdownMenuTrigger>
   245→   245→              <DropdownMenuContent align="end" className="w-44">
   246→   246→                <DropdownMenuLabel>{t("actions")}</DropdownMenuLabel>
   247→   247→                <DropdownMenuSeparator />
   248→   248→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "VERIFIED" })}>{t("actVerify")}</DropdownMenuItem>
   249→   249→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "PENDING" })}>{t("actSetPending")}</DropdownMenuItem>
   250→   250→                <DropdownMenuItem onClick={() => patch(tt.id, { verification: "REJECTED" })} className="text-rose-600 focus:text-rose-700">
   251→   251→                  {t("actReject")}
   252→   252→                </DropdownMenuItem>
   253→   253→                <DropdownMenuSeparator />

   265→   265→            </div>
   266→   266→            <div className="rounded-lg bg-slate-50 px-3 py-2">
   267→   267→              <p className="text-[10px] text-slate-400">{t("kpiListings")}</p>

   302→   302→
   303→   303→  const patch = async (id: string, body: Record<string, string | boolean>) => {
   304→   304→    setBusyId(id);

   306→   306→      await apiSend(`/api/admin/vehicles/${id}`, "PATCH", body);
   307→   307→      toast.success(t("toastUpdated"));
   308→   308→      refetch();
   309→   309→    } catch (e) {
   310→   310→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));

   312→   312→      setBusyId(null);
   313→   313→    }
   314→   314→  };

   333→   333→              )}
   334→   334→            >
   335→   335→              {t(`vst${c}` as never)}

   341→   341→      {loading && !data ? (
   342→   342→        <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>

   358→   358→                ) : (
   359→   359→                  <Car className="mx-auto h-6 w-6 text-slate-300" aria-hidden />

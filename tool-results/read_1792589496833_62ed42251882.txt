    64→/** Builds the API query params (advanced filters serialize arrays comma-joined). */
    65→function buildParams(f: BrowseFilters, sort: SortOption, page: number): Record<string, string | number> {
    66→  const params: Record<string, string | number> = { sort, limit: 12 };
    67→  if (page > 0) params.cursor = page;
    68→  if (f.q.trim()) params.q = f.q.trim();
    69→  if (f.make) params.make = f.make;
    70→  if (f.bodyType.length) params.bodyType = f.bodyType.join(",");
    71→  if (f.fuel.length) params.fuel = f.fuel.join(",");
    72→  if (f.transmission.length) params.transmission = f.transmission.join(",");
    73→  if (f.drivetrain.length) params.drivetrain = f.drivetrain.join(",");
    74→  if (f.condition) params.condition = f.condition;
    75→  if (f.city) params.city = f.city;
    76→  if (f.sellerType) params.sellerType = f.sellerType;
    77→  if (f.sellerVerified) params.sellerVerified = "true";
    78→  if (f.featured) params.featured = "true";
    79→  if (f.hasReport) params.hasReport = "true";
    80→  if (f.priceMin) params.priceMin = Number(f.priceMin);
    81→  if (f.priceMax) params.priceMax = Number(f.priceMax);
    82→  if (f.yearMin) params.yearMin = Number(f.yearMin);
    83→  if (f.yearMax) params.yearMax = Number(f.yearMax);
    84→  if (f.mileageMin) params.mileageMin = Number(f.mileageMin);
    85→  if (f.mileageMax) params.mileageMax = Number(f.mileageMax);
    86→  return params;
    87→}

    89→/** Toggle one value inside a multi-value filter array. */
    90→function toggleValue(f: BrowseFilters, field: "bodyType" | "fuel" | "transmission" | "drivetrain", v: string): BrowseFilters {
    91→  const cur = f[field];
    92→  return { ...f, [field]: cur.includes(v) ? cur.filter((x) => x !== v) : [...cur, v] };
    93→}

   421→
   422→  const removeChip = (key: string) => {
   423→    const next: BrowseFilters = { ...applied };
   424→    if (key.includes(":")) {
   425→      const [field, val] = key.split(":");
   426→      const f = field as "bodyType" | "fuel" | "transmission" | "drivetrain";
   427→      next[f] = next[f].filter((x) => x !== val);
   428→    } else if (key === "sellerVerified" || key === "featured" || key === "hasReport") {
   429→      next[key] = false;
   430→    } else {
   431→      (next as unknown as Record<string, string>)[key] = "";
   432→    }
   433→    applyFilters(next);

   435→

   437→    if (!saveName.trim()) return;
   438→    setSavingSearch(true);
   439→    try {
   440→      await apiSend("/api/account/saved-searches", "POST", { name: saveName.trim(), query: saveQuery });
   441→      toast.success(t("toastSearchSaved"));
   442→      setSaveOpen(false);
   443→      setSaveName("");
   444→    } catch (e) {
   445→      toast.error(errorText(e));
   446→    } finally {
   447→      setSavingSearch(false);
   448→    }
   449→  };
   450→
   451→  const items = data?.items ?? [];
   452→  const activeFilterCount = chips.length;
   453→
   454→  const advButton = (onClick: () => void) => (
   455→    <Button variant="outline" size="sm" onClick={onClick} aria-expanded={advOpen} aria-label={t("advFilters")}>
   456→      <SlidersHorizontal className="h-4 w-4 mr-1.5" aria-hidden />
   457→      {t("advFilters")}
   458→      {activeFilterCount > 0 && (
   459→        <span className="ml-1.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-amber-500 px-1 text-[11px] font-bold text-slate-950">
   460→          {activeFilterCount}
   461→        </span>
   462→      )}
   463→    </Button>
   464→  );
   465→
   466→  return (
   467→    <div className="mx-auto max-w-7xl px-4 py-6 pb-24 md:pb-10">
   468→      <div className="flex flex-wrap items-center justify-between gap-2">
   469→        <h1 className="text-xl sm:text-2xl font-bold text-slate-900">{t("navBrowse")}</h1>
   470→        <div className="flex flex-wrap items-center gap-2">
   471→          {canSaveSearch && (
   472→            <Button variant="outline" size="sm" onClick={() => setSaveOpen(true)} aria-label={t("saveSearchBtn")}>
   473→              <BookmarkPlus className="h-4 w-4 mr-1.5" aria-hidden />
   474→              {t("saveSearchBtn")}
   475→            </Button>
   476→          )}
   477→          {/* Mobile advanced sheet trigger */}
   478→          <div className="lg:hidden">{advButton(() => setAdvSheetOpen(true))}</div>
   479→          {/* Desktop advanced toggle */}
   480→          <div className="hidden lg:block">{advButton(() => setAdvOpen((o) => !o))}</div>
   481→
   482→          {/* Mobile quick filter sheet */}
   483→          <Sheet open={sheetOpen} onOpenChange={setSheetOpen}>
   484→            <SheetTrigger asChild>
   485→              <Button variant="outline" size="sm" className="md:hidden">
   486→                {t("filters")}
   487→                {activeFilterCount > 0 && (
   488→                  <span className="ml-1.5 flex h-5 w-5 items-center justify-center rounded-full bg-amber-500 text-[11px] font-bold text-slate-950">
   489→                    {activeFilterCount}
   490→                  </span>
   491→                )}
   492→              </Button>
   493→            </SheetTrigger>
   494→            <SheetContent side="bottom" className="h-[85vh] overflow-y-auto kl-scroll">
   495→              <SheetHeader>
   496→                <SheetTitle>{t("filters")}</SheetTitle>
   497→              </SheetHeader>
   498→              <div className="pb-6">
   499→                <FiltersPanel filters={filters} setFilters={setFilters} onClear={clearAll} />
   500→                <Button onClick={applyFromSheet} className="mt-4 w-full bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   501→                  <Search className="h-4 w-4 mr-2" aria-hidden />
   502→                  {t("applyFilters")}
   503→                </Button>
   504→              </div>
   505→            </SheetContent>
   506→          </Sheet>
   507→
   508→          <Select value={sort} onValueChange={(v) => { setSort(v as SortOption); setPage(0); }}>
   509→            <SelectTrigger className="h-9 w-[132px] text-sm sm:w-[180px]" aria-label={t("sortBy")}>
   510→              <SelectValue />
   511→            </SelectTrigger>
   512→            <SelectContent>
   513→              <SelectItem value="newest">{t("sortNewest")}</SelectItem>
   514→              <SelectItem value="price_asc">{t("sortPriceAsc")}</SelectItem>
   515→              <SelectItem value="price_desc">{t("sortPriceDesc")}</SelectItem>
   516→              <SelectItem value="year_desc">{t("sortYearDesc")}</SelectItem>
   517→              <SelectItem value="mileage_asc">{t("sortMileageAsc")}</SelectItem>
   518→            </SelectContent>
   519→          </Select>
   520→        </div>
   521→      </div>
   522→
   523→      {/* Active filter chips */}
   524→      {chips.length > 0 && (
   525→        <div className="mt-3 flex flex-wrap items-center gap-1.5" aria-label={t("activeFiltersLabel")}>
   526→          {chips.map((c) => (
   527→            <span
   528→              key={c.key}
   529→              className="inline-flex items-center gap-1 rounded-full border border-amber-300 bg-amber-50 py-1 pl-2.5 pr-1.5 text-xs font-medium text-amber-800"
   530→            >
   531→              {c.label}
   532→              <button
   533→                onClick={() => removeChip(c.key)}
   534→                className="flex h-4 w-4 items-center justify-center rounded-full text-amber-600 hover:bg-amber-200 cursor-pointer"
   535→                aria-label={`${t("remove")} ${c.label}`}
   536→              >
   537→                <X className="h-3 w-3" aria-hidden />
   538→              </button>
   539→            </span>
   540→          ))}
   541→          <button
   542→            onClick={clearAll}
   543→            className="inline-flex items-center rounded-full border border-slate-200 bg-white px-2.5 py-1 text-xs font-medium text-slate-500 hover:border-slate-300 hover:text-slate-700 cursor-pointer"
   544→          >
   545→            {t("clearFilters")}
   546→          </button>
   547→        </div>
   548→      )}
   549→
   550→      {/* Desktop advanced panel (collapsible card, spans full width) */}
   551→      {advOpen && (
   552→        <Card className="mt-4 hidden lg:block">
   553→          <CardContent className="p-5">
   554→            <p className="mb-4 text-sm font-semibold text-slate-900">{t("advFilters")}</p>
   555→            <AdvancedFields filters={filters} setFilters={applyFilters} onClear={clearAll} />
   556→            <Button
   557→              onClick={() => { setApplied(filters); setPage(0); setAdvOpen(false); }}
   558→              className="mt-4 bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold"
   559→            >
   560→              <Search className="h-4 w-4 mr-2" aria-hidden />
   561→              {t("applyFilters")}
   562→            </Button>
   563→          </CardContent>
   564→        </Card>
   565→      )}
   566→
   567→      <div className="mt-4 grid gap-6 lg:grid-cols-[260px_1fr]">
   568→        {/* Desktop quick filters */}
   569→        <aside className="hidden lg:block">
   570→          <div className="sticky top-20 rounded-xl border border-slate-200 bg-white p-4 max-h-[calc(100vh-6rem)] overflow-y-auto kl-scroll">
   571→            <p className="mb-4 text-sm font-semibold text-slate-900">{t("filters")}</p>
   572→            <FiltersPanel filters={filters} setFilters={applyFilters} onClear={clearAll} />
   573→          </div>
   574→        </aside>
   575→
   576→        {/* Results */}
   577→        <section aria-live="polite">
   578→          <p className="text-sm text-slate-500 mb-3">
   579→            {loading && page === 0 ? "…" : t("resultsCount", { n: data?.total ?? 0 })}
   580→          </p>
   581→          {error ? (
   582→            <div className="rounded-xl border border-rose-200 bg-rose-50 p-6 text-center text-sm text-rose-700">
   583→              {error}
   584→            </div>
   585→          ) : items.length === 0 && !loading ? (
   586→            <div className="rounded-xl border border-dashed border-slate-300 bg-slate-50 p-10 text-center">
   587→              <Search className="mx-auto h-8 w-8 text-slate-300" aria-hidden />
   588→              <p className="mt-3 font-medium text-slate-700">{t("noResults")}</p>
   589→              <p className="mt-1 text-sm text-slate-500">{t("noResultsHint")}</p>
   590→              <Button variant="outline" size="sm" onClick={clearAll} className="mt-4">
   591→                {t("clearFilters")}
   592→              </Button>
   593→            </div>
   594→          ) : (
   595→            <>
   596→              <div className="grid gap-4 grid-cols-1 sm:grid-cols-2 xl:grid-cols-3">
   597→                {loading && page === 0
   598→                  ? Array.from({ length: 6 }).map((_, i) => <VehicleCardSkeleton key={i} />)
   599→                  : items.map((v) => <VehicleCard key={v.id} vehicle={v} />)}
   600→              </div>
   601→              {data?.nextCursor && (
   602→                <div className="mt-6 text-center">
   603→                  <Button variant="outline" onClick={() => setPage((p) => p + 1)} disabled={loading}>
   604→                    {loading ? t("loading") : t("loadMore")}
   605→                  </Button>
   606→                </div>
   607→              )}
   608→              {!data?.nextCursor && items.length > 0 && (
   609→                <p className="mt-6 text-center text-xs text-slate-400">
   610→                  — {items.length} / {data?.total ?? 0} —
   611→                </p>
   612→              )}
   613→            </>
   614→          )}
   615→        </section>
   616→      </div>
   617→
   618→      {/* Mobile advanced bottom sheet */}
   619→      <Sheet open={advSheetOpen} onOpenChange={setAdvSheetOpen}>
   620→        <SheetContent side="bottom" className="h-[85vh] overflow-y-auto kl-scroll">
   621→          <SheetHeader>
   622→            <SheetTitle>{t("advFilters")}</SheetTitle>
   623→          </SheetHeader>
   624→          <div className="pb-6">
   625→            <AdvancedFields filters={filters} setFilters={setFilters} onClear={clearAll} />
   626→            <Button onClick={applyFromAdvSheet} className="mt-4 w-full bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   627→              <Search className="h-4 w-4 mr-2" aria-hidden />
   628→              {t("applyFilters")}
   629→            </Button>
   630→          </div>
   631→        </SheetContent>
   632→      </Sheet>
   633→
   634→      {/* Save search dialog */}
   635→      <Dialog open={saveOpen} onOpenChange={setSaveOpen}>
   636→        <DialogContent className="sm:max-w-sm">
   637→          <DialogHeader>
   638→            <DialogTitle className="flex items-center gap-2">
   639→              <BookmarkPlus className="h-4 w-4 text-amber-600" aria-hidden />
   640→              {t("saveSearchTitle")}
   641→            </DialogTitle>
   642→          </DialogHeader>
   643→          <div className="flex flex-wrap gap-1.5">
   644→            {chips.length === 0 ? (
   645→              <p className="text-xs text-slate-400">{t("noResultsHint")}</p>
   646→            ) : (
   647→              chips.slice(0, 6).map((c) => (
   648→                <span key={c.key} className="rounded-full border border-slate-200 bg-slate-50 px-2.5 py-0.5 text-[11px] text-slate-600">
   649→                  {c.label}
   650→                </span>
   651→              ))
   652→            )}
   653→          </div>
   654→          <div>
   655→            <Label htmlFor="ss-name" className="text-xs text-slate-500">{t("saveSearchNameLabel")}</Label>
   656→            <Input id="ss-name" value={saveName} onChange={(e) => setSaveName(e.target.value)} className="mt-1.5 h-9" autoFocus />
   657→          </div>
   658→          <DialogFooter>
   659→            <Button variant="outline" size="sm" onClick={() => setSaveOpen(false)}>{t("cancel")}</Button>
   660→            <Button
   661→              size="sm" onClick={saveSearch} disabled={savingSearch || !saveName.trim()}
   662→              className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold"
   663→            >
   664→              {savingSearch && <Loader2 className="h-4 w-4 mr-2 animate-spin" aria-hidden />}
   665→              {t("saveSearchBtn")}
   666→            </Button>
   667→          </DialogFooter>
   668→        </DialogContent>
   669→      </Dialog>
   670→
   671→      {/* Back to home for mobile users landing deep */}
   672→      <div className="mt-8 text-center md:hidden">
   673→        <Button variant="ghost" size="sm" onClick={() => navigate({ name: "home" })} className="text-slate-500">
   674→          ← {t("navHome")}
   675→        </Button>
   676→      </div>
   677→    </div>
   678→  );
   679→}

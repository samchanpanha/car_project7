   199→function AdminTenants() {
   200→  const t = useT();
   201→  const { data, loading, error, refetch } = useFetch<AdminTenantDTO[]>("/api/admin/tenants");
   202→  const [busyId, setBusyId] = useState<string | null>(null);
   203→
   204→  const patch = async (id: string, body: Record<string, string>) => {
   205→    setBusyId(id);
   206→    try {
   207→      await apiSend(`/api/admin/tenants/${id}`, "PATCH", body);
   208→      toast.success(t("toastUpdated"));
   209→      refetch();
   210→    } catch (e) {
   211→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   212→    } finally {
   213→      setBusyId(null);
   214→    }
   215→  };
   216→
   217→  if (loading && !data) {
   218→    return <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}</div>;
   219→  }
   220→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   221→  if (!data || data.length === 0) return <EmptyState icon={Store} text={t("noResults")} />;
   222→
   223→  return (
   224→    <div className="space-y-3">
   225→      {data.map((tt) => (
   226→        <div key={tt.id} className="rounded-xl border border-slate-200 bg-white p-4">

   287→function AdminListings() {
   288→  const t = useT();
   289→  const navigate = useAppStore((s) => s.navigate);
   290→  const [status, setStatus] = useState<string>("PENDING_REVIEW");
   291→  const [q, setQ] = useState("");
   292→  const [debQ, setDebQ] = useState("");
   293→  const [busyId, setBusyId] = useState<string | null>(null);
   294→
   295→  useEffect(() => {
   296→    const h = setTimeout(() => setDebQ(q.trim()), 350);
   297→    return () => clearTimeout(h);
   298→  }, [q]);
   299→
   300→  const url = `/api/admin/vehicles${qs({ status: status === "ALL" ? undefined : status, q: debQ || undefined })}`;
   301→  const { data, loading, error, refetch } = useFetch<AdminVehicleRowDTO[]>(url);
   302→
   303→  const patch = async (id: string, body: Record<string, string | boolean>) => {
   304→    setBusyId(id);
   305→    try {
   306→      await apiSend(`/api/admin/vehicles/${id}`, "PATCH", body);
   307→      toast.success(t("toastUpdated"));
   308→      refetch();
   309→    } catch (e) {
   310→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   311→    } finally {
   312→      setBusyId(null);
   313→    }
   314→  };
   315→
   316→  const chips = ["ALL", ...VEHICLE_STATUSES];
   317→
   318→  return (
   319→    <div className="space-y-4">
   320→      <div className="flex flex-wrap items-center gap-2">
   321→        <div className="flex w-full items-center gap-2 sm:w-64">
   322→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder={`${t("searchButton")}…`} className="h-9" aria-label={t("searchButton")} />
   323→        </div>
   324→        <div className="flex flex-wrap gap-1.5">
   325→          {chips.map((c) => (
   326→            <button
   327→              key={c}
   328→              onClick={() => setStatus(c)}
   329→              aria-pressed={status === c}
   330→              className={cn(
   331→                "rounded-full border px-3 py-1.5 text-xs font-medium transition-colors cursor-pointer min-h-[32px]",
   332→                status === c ? "border-amber-500 bg-amber-50 text-amber-800" : "border-slate-200 bg-white text-slate-600 hover:border-slate-300"
   333→              )}
   334→            >
   335→              {t(`vst${c}` as never)}
   336→            </button>
   337→          ))}
   338→        </div>
   339→      </div>
   340→
   341→      {loading && !data ? (
   342→        <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 w-full rounded-xl" />)}</div>
   343→      ) : error && !data ? (
   344→        <FetchErrorState message={error} onRetry={refetch} />
   345→      ) : !data || data.length === 0 ? (
   346→        <EmptyState icon={Car} text={t("noResults")} />

   348→        <div className="space-y-2.5">

   575→function AdminSettings() {
   576→  const t = useT();
   577→  const { data, loading, error, refetch } = useFetch<PlatformSettingsDTO>("/api/admin/settings");
   578→  const [form, setForm] = useState<PlatformSettingsDTO | null>(null);
   579→  const [saving, setSaving] = useState(false);
   580→
   581→  useEffect(() => {
   582→    if (data) setForm(data);
   583→  }, [data]);
   584→
   585→  const save = async () => {
   586→    if (!form) return;
   587→    setSaving(true);
   588→    try {
   589→      const saved = await apiSend<PlatformSettingsDTO>("/api/admin/settings", "PUT", {
   590→        commissionPct: Number(form.commissionPct) || 0,
   591→        featuredListingPriceUsd: Number(form.featuredListingPriceUsd) || 0,
   592→        autoApproveListings: form.autoApproveListings,
   593→        maintenanceMode: form.maintenanceMode,
   594→      });
   595→      setForm(saved);
   596→      toast.success(t("toastUpdated"));
   597→      refetch();
   598→    } catch (e) {
   599→      toast.error(isForbidden(e) ? t("toastForbidden") : errorText(e));
   600→    } finally {
   601→      setSaving(false);
   602→    }
   603→  };
   604→
   605→  if (loading && !data) return <Skeleton className="h-72 w-full rounded-xl" />;
   606→  if (error && !data) return <FetchErrorState message={error} onRetry={refetch} />;
   607→  if (!form) return null;
   608→
   609→  return (
   610→    <SectionCard title={t("platformSettings")}>
   611→      {form.maintenanceMode && (
   612→        <Alert className="mb-4 border-amber-300 bg-amber-50 text-amber-900">
   613→          <TriangleAlert className="h-4 w-4" aria-hidden />
   614→          <AlertTitle>{t("setMaintenance")}</AlertTitle>
   615→          <AlertDescription>{t("maintenanceWarning")}</AlertDescription>
   616→        </Alert>
   617→      )}
   618→      <div className="grid gap-4 sm:grid-cols-2">
   619→        <div>
   620→          <Label htmlFor="adm-comm" className="text-xs text-slate-500">{t("setCommission")}</Label>
   621→          <Input
   622→            id="adm-comm" type="number" min={0} max={100} step={0.1} className="mt-1.5 h-9"
   623→            value={form.commissionPct}
   624→            onChange={(e) => setForm({ ...form, commissionPct: Number(e.target.value) })}
   625→          />
   626→        </div>
   627→        <div>
   628→          <Label htmlFor="adm-feat" className="text-xs text-slate-500">{t("setFeaturedPrice")}</Label>
   629→          <Input
   630→            id="adm-feat" type="number" min={0} className="mt-1.5 h-9"
   631→            value={form.featuredListingPriceUsd}
   632→            onChange={(e) => setForm({ ...form, featuredListingPriceUsd: Number(e.target.value) })}
   633→          />
   634→        </div>
   635→        <div className="flex items-center justify-between rounded-lg border border-slate-200 p-3">
   636→          <Label htmlFor="adm-auto" className="text-xs font-medium text-slate-600 cursor-pointer">{t("setAutoApprove")}</Label>
   637→          <Switch id="adm-auto" checked={form.autoApproveListings} onCheckedChange={(v) => setForm({ ...form, autoApproveListings: v })} />
   638→        </div>
   639→        <div className="flex items-center justify-between rounded-lg border border-slate-200 p-3">
   640→          <Label htmlFor="adm-maint" className="text-xs font-medium text-slate-600 cursor-pointer">{t("setMaintenance")}</Label>
   641→          <Switch id="adm-maint" checked={form.maintenanceMode} onCheckedChange={(v) => setForm({ ...form, maintenanceMode: v })} />
   642→        </div>
   643→      </div>
   644→      <Button onClick={save} disabled={saving} className="mt-4 bg-amber-500 hover:bg-amber-400 text-slate-950 font-semibold">
   645→        {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" aria-hidden />}
   646→        {t("saveChanges")}
   647→      </Button>
   648→    </SectionCard>
   649→  );
   650→}
